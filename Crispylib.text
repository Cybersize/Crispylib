local function Create(class, props)
    local obj = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then obj[k] = v end
    end
    if props and props.Parent then obj.Parent = props.Parent end
    return obj
end

local function Tween(obj, goals, ti)
    TweenService:Create(obj, ti or TI_MID, goals):Play()
end

local function Round(f, r)
    Create("UICorner", { CornerRadius = UDim.new(0, r or 8), Parent = f })
end

local function Stroke(f, color, thickness)
    return Create("UIStroke", {
        Color = color or Theme.Border, Thickness = thickness or 1, Parent = f,
    })
end

local function AddPadding(f, top, bot, left, right)
    Create("UIPadding", {
        PaddingTop    = UDim.new(0, top   or 0),
        PaddingBottom = UDim.new(0, bot   or 0),
        PaddingLeft   = UDim.new(0, left  or 0),
        PaddingRight  = UDim.new(0, right or 0),
        Parent = f,
    })
end

local function ListLayout(f, dir, spacing)
    Create("UIListLayout", {
        FillDirection = dir or Enum.FillDirection.Vertical,
        SortOrder     = Enum.SortOrder.LayoutOrder,
        Padding       = UDim.new(0, spacing or 0),
        Parent        = f,
    })
end

local function ScrollFrame(parent, size, pos)
    return Create("ScrollingFrame", {
        Size                   = size or UDim2.new(1, 0, 1, 0),
        Position               = pos  or UDim2.new(0, 0, 0, 0),
        BackgroundTransparency = 1,
        ScrollBarThickness     = 3,
        ScrollBarImageColor3   = Theme.ScrollbarThumb,
        BorderSizePixel        = 0,
        CanvasSize             = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize    = Enum.AutomaticSize.Y,
        Parent                 = parent,
    })
end

local function MakeDraggable(handle, frame)
    local dragging, ds, sp
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true; ds = i.Position; sp = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then
            local d = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
end

-- Hover highlight for rows
local function AddRowHover(row)
    local bg = Create("Frame", {
        Name = "HoverBg", Size = UDim2.new(1, 0, 1, 0), Position = UDim2.new(0,0,0,0),
        BackgroundColor3 = Theme.RowHover, BackgroundTransparency = 1,
        BorderSizePixel = 0, ZIndex = row.ZIndex, Parent = row,
    })
    -- detect button or any clickable child; listen on row itself
    row.MouseEnter:Connect(function()
        Tween(bg, { BackgroundTransparency = 0.72 }, TI_FAST)
    end)
    row.MouseLeave:Connect(function()
        Tween(bg, { BackgroundTransparency = 1 }, TI_FAST)
    end)
    return bg
end

-- Focus glow helper for input/dropdown frames
local function AddFocusStroke(frame, box)
    local stroke = Stroke(frame, Theme.InputBorder, 1)
    box.Focused:Connect(function()
        Tween(stroke, { Color = Theme.FocusBorder, Thickness = 1.5 }, TI_FAST)
    end)
    box.FocusLost:Connect(function()
        Tween(stroke, { Color = Theme.InputBorder, Thickness = 1 }, TI_FAST)
    end)
    return stroke
end

local Crispylib            = {}
Crispylib.__index          = Crispylib
Crispylib._flags           = {}
Crispylib._flagElements    = {}
Crispylib._configName      = "Crispylib"

local _hasWritefile  = (type(writefile)   == "function")
local _hasReadfile   = (type(readfile)    == "function")
local _hasListfiles  = (type(listfiles)   == "function")
local _hasIsfile     = (type(isfile)      == "function")
local _hasMakefolder = (type(makefolder)  == "function")

local function SerialiseValue(v)
    if typeof(v) == "Color3" then return { __type="Color3", r=v.R, g=v.G, b=v.B }
    elseif type(v) == "table" then
        local o={}; for i,x in ipairs(v) do o[i]=SerialiseValue(x) end; return o
    end; return v
end

local function DeserialiseValue(v)
    if type(v) == "table" then
        if v.__type == "Color3" then return Color3.new(v.r, v.g, v.b) end
        local o={}; for i,x in ipairs(v) do o[i]=DeserialiseValue(x) end; return o
    end; return v
end

local function ConfigFolder() return "Crispylib/" .. Crispylib._configName end
local function ProfilePath(n) return ConfigFolder() .. "/" .. tostring(n) .. ".json" end
local function EnsureFolder()
    if _hasMakefolder then pcall(makefolder,"Crispylib"); pcall(makefolder,ConfigFolder()) end
end

local function RegisterFlag(flag, getF, setF)
    if flag then Crispylib._flagElements[flag] = { Get=getF, Set=setF } end
end

Crispylib.Config = {}

function Crispylib.Config.Snapshot()
    local s={};for f,e in pairs(Crispylib._flagElements) do s[f]=SerialiseValue(e.Get()) end; return s
end

function Crispylib.Config.Apply(snap)
    for f,raw in pairs(snap) do
        local e=Crispylib._flagElements[f]; if e then pcall(e.Set,DeserialiseValue(raw)) end
    end
end

function Crispylib.Config.Export()
    local ok,j=pcall(function() return HttpService:JSONEncode(Crispylib.Config.Snapshot()) end)
    return ok and j or "{}"
end

function Crispylib.Config.Import(json)
    local ok,snap=pcall(function() return HttpService:JSONDecode(json) end)
    if ok and type(snap)=="table" then Crispylib.Config.Apply(snap); return true end; return false
end

function Crispylib.Config.Save(name)
    name=name or "default"
    if not _hasWritefile then
        warn("[Crispylib] writefile unavailable — in-memory only")
        Crispylib._inMemory=Crispylib._inMemory or {}; Crispylib._inMemory[name]=Crispylib.Config.Snapshot()
        return true
    end
    EnsureFolder(); return pcall(writefile,ProfilePath(name),Crispylib.Config.Export())
end

function Crispylib.Config.Load(name)
    name=name or "default"
    if not _hasReadfile then
        if Crispylib._inMemory and Crispylib._inMemory[name] then
            Crispylib.Config.Apply(Crispylib._inMemory[name]); return true
        end; return false
    end
    local path=ProfilePath(name)
    if _hasIsfile and not isfile(path) then return false end
    local ok,json=pcall(readfile,path); if not ok then return false end
    return Crispylib.Config.Import(json)
end

function Crispylib.Config.List()
    if not _hasListfiles then
        local l={}; if Crispylib._inMemory then for k in pairs(Crispylib._inMemory) do l[#l+1]=k end end; return l
    end
    EnsureFolder(); local ok,files=pcall(listfiles,ConfigFolder()); if not ok then return {} end
    local names={}
    for _,p in ipairs(files) do local n=p:match("([^/\\]+)%.json$"); if n then names[#names+1]=n end end
    return names
end

function Crispylib.Config.Delete(name)
    if not name then return false end
    if not _hasWritefile then
        if Crispylib._inMemory then Crispylib._inMemory[name]=nil end; return true
    end
    local path=ProfilePath(name)
    if _hasIsfile and not isfile(path) then return false end
    if type(delfile)=="function" then pcall(delfile,path) else pcall(writefile,path,"{}") end
    return true
end

function Crispylib.CreateLoadingScreen(cfg)

    local title   = cfg.Title       or "Loading"
    local sub     = cfg.Subtitle    or ""
    local logoId  = cfg.LogoId      or ""
    local minTime = cfg.MinimumTime or 0
    local startT  = os.clock()

    local sg = Create("ScreenGui", {
        Name="Crispylib_Loader", ResetOnSpawn=false,
        ZIndexBehavior=Enum.ZIndexBehavior.Sibling,
        IgnoreGuiInset=true, DisplayOrder=2000,
    })
    pcall(function() sg.Parent=CoreGui end)
    if not sg.Parent then sg.Parent=LocalPlayer:WaitForChild("PlayerGui") end

    local bg = Create("Frame", {
        Size=UDim2.new(1,0,1,0), BackgroundColor3=Theme.LoaderBg,
        BorderSizePixel=0, Parent=sg,
    })
    Create("UIGradient", {
        Color=ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(20,20,24)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(12,12,16)),
        }),
        Rotation=140, Parent=bg,
    })

    -- Subtle grid overlay
    Create("ImageLabel", {
        Size=UDim2.new(1,0,1,0), BackgroundTransparency=1,
        Image="rbxassetid://7072725342", ImageTransparency=0.92,
        ScaleType=Enum.ScaleType.Tile, TileSize=UDim2.new(0,64,0,64),
        Parent=bg,
    })

    local center = Create("Frame", {
        Size=UDim2.new(0,360,0,280), Position=UDim2.new(0.5,-180,0.5,-140),
        BackgroundTransparency=1, Parent=bg,
    })

    local yOff = 0
    if logoId ~= "" then
        local logoFrame = Create("Frame", {
            Size=UDim2.new(0,72,0,72), Position=UDim2.new(0.5,-36,0,0),
            BackgroundColor3=Color3.fromRGB(30,30,36), BorderSizePixel=0, Parent=center,
        })
        Round(logoFrame, 18)
        Create("ImageLabel", {
            Size=UDim2.new(0,56,0,56), Position=UDim2.new(0.5,-28,0.5,-28),
            BackgroundTransparency=1, Image=logoId, Parent=logoFrame,
        })
        yOff = 84
    end

    Create("TextLabel", {
        Size=UDim2.new(1,0,0,36), Position=UDim2.new(0,0,0,yOff),
        BackgroundTransparency=1, Text=title,
        TextColor3=Theme.TitleText, TextSize=26, Font=Enum.Font.GothamBold,
        TextXAlignment=Enum.TextXAlignment.Center, Parent=center,
    })

    local subLabel = Create("TextLabel", {
        Size=UDim2.new(1,0,0,18), Position=UDim2.new(0,0,0,yOff+40),
        BackgroundTransparency=1, Text=sub,
        TextColor3=Theme.SubtitleText, TextSize=13, Font=Enum.Font.Gotham,
        TextXAlignment=Enum.TextXAlignment.Center, Parent=center,
    })

    local barBg = Create("Frame", {
        Size=UDim2.new(1,0,0,5), Position=UDim2.new(0,0,0,yOff+74),
        BackgroundColor3=Theme.LoaderBarBg, BorderSizePixel=0, Parent=center,
    })
    Round(barBg,3)
    local barFill = Create("Frame", {
        Size=UDim2.new(0,0,1,0), BackgroundColor3=Theme.LoaderBarFill,
        BorderSizePixel=0, Parent=barBg,
    })
    Round(barFill,3)

    local statusLabel = Create("TextLabel", {
        Size=UDim2.new(1,0,0,16), Position=UDim2.new(0,0,0,yOff+88),
        BackgroundTransparency=1, Text="",
        TextColor3=Theme.DescText, TextSize=11, Font=Enum.Font.Gotham,
        TextXAlignment=Enum.TextXAlignment.Center, Parent=center,
    })

    local taskList = Create("Frame", {
        Size=UDim2.new(1,0,0,110), Position=UDim2.new(0,0,0,yOff+112),
        BackgroundTransparency=1, ClipsDescendants=true, Parent=center,
    })
    ListLayout(taskList, Enum.FillDirection.Vertical, 5)

    local L = {}
    local progress = 0

    function L:SetProgress(pct)
        progress = math.clamp(pct,0,1)
        Tween(barFill, {Size=UDim2.new(progress,0,1,0)},
            TweenInfo.new(0.3,Enum.EasingStyle.Quad,Enum.EasingDirection.Out))
    end
    function L:SetStatus(text) statusLabel.Text=tostring(text or "") end
    function L:AddTask(text)
        local lbl=Create("TextLabel", {
            Size=UDim2.new(1,0,0,15), BackgroundTransparency=1,
            Text="✓  "..tostring(text), TextColor3=Theme.DescText,
            TextSize=11, Font=Enum.Font.Gotham,
            TextXAlignment=Enum.TextXAlignment.Center, TextTransparency=0.3, Parent=taskList,
        })
        for _,c in ipairs(taskList:GetChildren()) do
            if c:IsA("TextLabel") then
                Tween(c, {TextTransparency=0.65}, TI_FAST)
            end
        end
        lbl.TextTransparency=0.1
    end
    function L:Finish(cb)
        local elapsed=os.clock()-startT
        local rem=math.max(0,minTime-elapsed)
        task.delay(rem, function()
            self:SetProgress(1); task.wait(0.28)
            Tween(bg,{BackgroundTransparency=1},TI_MID)
            for _,d in ipairs(bg:GetDescendants()) do
                if d:IsA("TextLabel") or d:IsA("ImageLabel") then
                    pcall(function() Tween(d,{TextTransparency=1,ImageTransparency=1},TI_MID) end)
                elseif d:IsA("Frame") and d~=bg then
                    pcall(function() Tween(d,{BackgroundTransparency=1},TI_MID) end)
                end
            end
            task.delay(0.35, function() sg:Destroy(); if cb then pcall(cb) end end)
        end)
    end

    task.spawn(function()
        while sg and sg.Parent do
            Tween(barFill,{BackgroundColor3=Color3.fromRGB(48,168,255)},
                TweenInfo.new(0.85,Enum.EasingStyle.Sine,Enum.EasingDirection.InOut))
            task.wait(1)
            Tween(barFill,{BackgroundColor3=Theme.LoaderBarFill},
                TweenInfo.new(0.85,Enum.EasingStyle.Sine,Enum.EasingDirection.InOut))
            task.wait(1)
        end
    end)

    return L
end

-- ────────────────────────────────────────────────────────────────────────────
--  CREATE WINDOW
-- ────────────────────────────────────────────────────────────────────────────
function Crispylib.CreateWindow(cfg)
    cfg = cfg or {}
    local title      = cfg.Title      or "Xenon Hub"
    local subtitle   = cfg.Subtitle   or ""
    local configName = cfg.ConfigName
    local autoLoad   = cfg.AutoLoad   or false

    if configName then Crispylib._configName = configName end

    local sg = Create("ScreenGui", {
        Name = "Crispylib_"..title:gsub("%s+",""),
        ResetOnSpawn=false, ZIndexBehavior=Enum.ZIndexBehavior.Sibling,
        IgnoreGuiInset=true, DisplayOrder=999,
    })
    pcall(function() sg.Parent=CoreGui end)
    if not sg.Parent then sg.Parent=LocalPlayer:WaitForChild("PlayerGui") end

    -- ── WINDOW ───────────────────────────────────────────────────────────
    -- Intro scale animation
    local window = Create("Frame", {
        Name="Window",
        Size=UDim2.new(0,WIN_W,0,WIN_H),
        Position=UDim2.new(0.5,-WIN_W/2,0.5,-WIN_H/2),
        BackgroundColor3=Theme.WindowBg,
        BorderSizePixel=0, Parent=sg,
    })
    Round(window, 13)
    Stroke(window, Color3.fromRGB(52,52,60), 1)

    -- Multi-layer shadow
    for i=1,3 do
        Create("Frame", {
            Size=UDim2.new(1,8+i*8,1,8+i*8),
            Position=UDim2.new(0,-4-i*4,0,4+i*3),
            BackgroundColor3=Color3.fromRGB(0,0,0),
            BackgroundTransparency=0.55+i*0.12,
            ZIndex=window.ZIndex-1, BorderSizePixel=0, Parent=window,
        })
        Round(window:GetChildren()[#window:GetChildren()], 16+i*2)
    end

    -- Animate in
    window.Size=UDim2.new(0,WIN_W*0.92,0,WIN_H*0.92)
    window.BackgroundTransparency=0.6
    Tween(window,{Size=UDim2.new(0,WIN_W,0,WIN_H),BackgroundTransparency=0},TI_EASE)

    -- ── TITLE BAR ────────────────────────────────────────────────────────
    local titleBar = Create("Frame", {
        Name="TitleBar",
        Size=UDim2.new(1,0,0,TITLEBAR_H),
        BackgroundColor3=Theme.TitleBarBg,
        BorderSizePixel=0, ZIndex=2, Parent=window,
    })
    Round(titleBar, 13)

    -- Bottom edge fill to hide bottom-rounded corners
    Create("Frame", {
        Size=UDim2.new(1,0,0,14), Position=UDim2.new(0,0,1,-14),
        BackgroundColor3=Theme.TitleBarBg, BorderSizePixel=0, ZIndex=2, Parent=titleBar,
    })

    -- Subtle bottom separator
    Create("Frame", {
        Size=UDim2.new(1,0,0,1), Position=UDim2.new(0,0,1,-1),
        BackgroundColor3=Theme.Separator, BorderSizePixel=0, Parent=titleBar,
    })

    -- Traffic lights
    local function TBtn(col, par)
        local f = Create("Frame", {
            Size=UDim2.new(0,13,0,13), BackgroundColor3=col,
            BorderSizePixel=0, Parent=par,
        })
        Round(f,7); return f
    end
    local btns = Create("Frame", {
        Size=UDim2.new(0,60,0,14), Position=UDim2.new(0,14,0.5,-7),
        BackgroundTransparency=1, Parent=titleBar,
    })
    ListLayout(btns, Enum.FillDirection.Horizontal, 8)
    local closeBtn = TBtn(Theme.CloseBtn, btns)
    local minBtn   = TBtn(Theme.MinBtn,   btns)
    TBtn(Theme.MaxBtn, btns)

    -- Sidebar toggle (panel icon)
    local sideToggle = Create("TextButton", {
        Size=UDim2.new(0,28,0,28), Position=UDim2.new(0,88,0.5,-14),
        BackgroundColor3=Theme.RowHover, BackgroundTransparency=1,
        Text="⊟", TextColor3=Theme.SubtitleText,
        TextSize=16, Font=Enum.Font.GothamBold, BorderSizePixel=0,
        AutoButtonColor=false, Parent=titleBar,
    })
    Round(sideToggle, 6)
    sideToggle.MouseEnter:Connect(function()
        Tween(sideToggle,{BackgroundTransparency=0.5},TI_FAST)
    end)
    sideToggle.MouseLeave:Connect(function()
        Tween(sideToggle,{BackgroundTransparency=1},TI_FAST)
    end)

    -- Nav arrows
    local function NavBtn(txt, x)
        local b = Create("TextButton", {
            Size=UDim2.new(0,22,0,26), Position=UDim2.new(0,x,0.5,-13),
            BackgroundTransparency=1, Text=txt, TextColor3=Theme.SubtitleText,
            TextSize=20, Font=Enum.Font.GothamBold, BorderSizePixel=0, Parent=titleBar,
        })
        return b
    end
    NavBtn("‹", 124); NavBtn("›", 148)

    -- Center title
    local tFrame = Create("Frame", {
        Size=UDim2.new(0,220,1,0), Position=UDim2.new(0.5,-110,0,0),
        BackgroundTransparency=1, Parent=titleBar,
    })
    Create("TextLabel", {
        Size=UDim2.new(1,0,0,18), Position=UDim2.new(0,0,0.5,-19),
        BackgroundTransparency=1, Text=title,
        TextColor3=Theme.TitleText, TextSize=14, Font=Enum.Font.GothamBold,
        TextXAlignment=Enum.TextXAlignment.Center, Parent=tFrame,
    })
    Create("TextLabel", {
        Size=UDim2.new(1,0,0,14), Position=UDim2.new(0,0,0.5,1),
        BackgroundTransparency=1, Text=subtitle,
        TextColor3=Theme.SubtitleText, TextSize=11, Font=Enum.Font.Gotham,
        TextXAlignment=Enum.TextXAlignment.Center, Parent=tFrame,
    })

    -- Search bar
    local searchBg = Create("Frame", {
        Size=UDim2.new(0,155,0,28), Position=UDim2.new(1,-168,0.5,-14),
        BackgroundColor3=Theme.InputBg, BorderSizePixel=0, Parent=titleBar,
    })
    Round(searchBg, 8)
    local sStroke = Stroke(searchBg, Theme.InputBorder, 1)
    Create("TextLabel", {
        Size=UDim2.new(0,22,1,0), Position=UDim2.new(0,6,0,0),
        BackgroundTransparency=1, Text="⌕",
        TextColor3=Theme.SearchPlaceholder, TextSize=15, Font=Enum.Font.Gotham, Parent=searchBg,
    })
    local sBox = Create("TextBox", {
        Size=UDim2.new(1,-28,1,0), Position=UDim2.new(0,26,0,0),
        BackgroundTransparency=1, PlaceholderText="Search",
        PlaceholderColor3=Theme.SearchPlaceholder, Text="",
        TextColor3=Theme.LabelText, TextSize=12, Font=Enum.Font.Gotham,
        TextXAlignment=Enum.TextXAlignment.Left, ClearTextOnFocus=false, Parent=searchBg,
    })
    sBox.Focused:Connect(function() Tween(sStroke,{Color=Theme.FocusBorder},TI_FAST) end)
    sBox.FocusLost:Connect(function() Tween(sStroke,{Color=Theme.InputBorder},TI_FAST) end)

    -- ── SIDEBAR ──────────────────────────────────────────────────────────
    local sidebar = Create("Frame", {
        Name="Sidebar",
        Size=UDim2.new(0,SIDEBAR_W,1,-TITLEBAR_H),
        Position=UDim2.new(0,0,0,TITLEBAR_H),
        BackgroundColor3=Theme.SidebarBg, BorderSizePixel=0, Parent=window,
    })
    -- Right border
    Create("Frame", {
        Size=UDim2.new(0,1,1,0), Position=UDim2.new(1,-1,0,0),
        BackgroundColor3=Theme.Separator, BorderSizePixel=0, Parent=sidebar,
    })

    local sbScroll = ScrollFrame(sidebar)
    AddPadding(sbScroll, 12, 12, 0, 0)
    local sbList = Create("Frame", {
        Name="List", Size=UDim2.new(1,0,0,0), AutomaticSize=Enum.AutomaticSize.Y,
        BackgroundTransparency=1, Parent=sbScroll,
    })
    ListLayout(sbList, Enum.FillDirection.Vertical, 1)

    -- ── CONTENT ──────────────────────────────────────────────────────────
    local content = Create("Frame", {
        Name="Content",
        Size=UDim2.new(1,-SIDEBAR_W,1,-TITLEBAR_H),
        Position=UDim2.new(0,SIDEBAR_W,0,TITLEBAR_H),
        BackgroundColor3=Theme.ContentBg, BorderSizePixel=0, Parent=window,
    })

    MakeDraggable(titleBar, window)

    -- ── WINDOW CONTROLS ──────────────────────────────────────────────────
    local minimised = false
    closeBtn.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 then
            Tween(window,{Size=UDim2.new(0,WIN_W*0.9,0,WIN_H*0.9),BackgroundTransparency=1},TI_MID)
            task.wait(0.25); sg:Destroy()
        end
    end)
    minBtn.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 then
            minimised=not minimised
            Tween(window,{Size=UDim2.new(0,WIN_W,0,minimised and TITLEBAR_H or WIN_H)},TI_SLOW)
        end
    end)

    local sbVisible = true
    sideToggle.MouseButton1Click:Connect(function()
        sbVisible=not sbVisible
        if sbVisible then
            Tween(sidebar,{Size=UDim2.new(0,SIDEBAR_W,1,-TITLEBAR_H)},TI_SLOW)
            Tween(content,{
                Size=UDim2.new(1,-SIDEBAR_W,1,-TITLEBAR_H),
                Position=UDim2.new(0,SIDEBAR_W,0,TITLEBAR_H),
            },TI_SLOW)
        else
            Tween(sidebar,{Size=UDim2.new(0,0,1,-TITLEBAR_H)},TI_SLOW)
            Tween(content,{
                Size=UDim2.new(1,0,1,-TITLEBAR_H),
                Position=UDim2.new(0,0,0,TITLEBAR_H),
            },TI_SLOW)
        end
    end)

    -- ── WINDOW OBJECT ────────────────────────────────────────────────────
    local Win         = {}
    Win._tabs         = {}
    Win._activeTab    = nil
    Win._sbList       = sbList
    Win._content      = content
    Win._order        = 0

    -- ── SIDEBAR SECTION LABEL ─────────────────────────────────────────────
    function Win:AddSidebarSection(name)
        local lbl = Create("TextLabel", {
            Name="Sec_"..name, Size=UDim2.new(1,0,0,24),
            BackgroundTransparency=1, Text=name:upper(),
            TextColor3=Theme.SectionLabel, TextSize=10, Font=Enum.Font.GothamBold,
            TextXAlignment=Enum.TextXAlignment.Left,
            LayoutOrder=self._order, Parent=self._sbList,
        })
        AddPadding(lbl, 0, 0, 18, 0)
        self._order=self._order+1
    end

    -- ── ADD TAB ───────────────────────────────────────────────────────────
    function Win:AddTab(tabCfg)
        tabCfg=tabCfg or {}
        local tName=tabCfg.Name or "Tab"
        local tIcon=tabCfg.Icon or ""
        self._order=self._order+1

        -- Sidebar pill button
        local tabBtn = Create("TextButton", {
            Name="Tab_"..tName, Size=UDim2.new(1,-16,0,34),
            BackgroundColor3=Theme.SidebarBg, BackgroundTransparency=1,
            Text="", BorderSizePixel=0, LayoutOrder=self._order,
            Parent=self._sbList, AutoButtonColor=false,
        })
        Round(tabBtn, 8)

        -- Active left accent bar (hidden by default)
        local accentBar = Create("Frame", {
            Name="Accent", Size=UDim2.new(0,3,0,18),
            Position=UDim2.new(0,1,0.5,-9), AnchorPoint=Vector2.new(0,0),
            BackgroundColor3=Theme.Accent, BackgroundTransparency=1,
            BorderSizePixel=0, Parent=tabBtn,
        })
        Round(accentBar,2)

        -- Icon
        local iconLbl = Create("TextLabel", {
            Name="Icon", Size=UDim2.new(0,20,1,0), Position=UDim2.new(0,14,0,0),
            BackgroundTransparency=1, Text=tIcon,
            TextColor3=Theme.TabInactiveText, TextSize=14, Font=Enum.Font.Gotham,
            TextXAlignment=Enum.TextXAlignment.Left, Parent=tabBtn,
        })
        local nameLbl = Create("TextLabel", {
            Name="Label", Size=UDim2.new(1,-38,1,0), Position=UDim2.new(0,36,0,0),
            BackgroundTransparency=1, Text=tName,
            TextColor3=Theme.TabInactiveText, TextSize=13, Font=Enum.Font.Gotham,
            TextXAlignment=Enum.TextXAlignment.Left, Parent=tabBtn,
        })

        -- Tab content scroll
        local tabScroll = ScrollFrame(self._content)
        tabScroll.Visible=false
        AddPadding(tabScroll,2,16,0,0)

        local tabContent = Create("Frame", {
            Name="Content_"..tName, Size=UDim2.new(1,0,0,0),
            AutomaticSize=Enum.AutomaticSize.Y, BackgroundTransparency=1,
            Parent=tabScroll,
        })
        ListLayout(tabContent, Enum.FillDirection.Vertical, 0)

        local Tab     = {}
        Tab._content  = tabContent
        Tab._scroll   = tabScroll
        Tab._btn      = tabBtn
        Tab._win      = self
        Tab._order    = 0

        local function Activate()
            if self._activeTab and self._activeTab~=Tab then
                local prev=self._activeTab
                Tween(prev._btn,{BackgroundColor3=Theme.SidebarBg,BackgroundTransparency=1},TI_MID)
                prev._btn:FindFirstChild("Accent").BackgroundTransparency=1
                prev._btn:FindFirstChild("Icon").TextColor3=Theme.TabInactiveText
                prev._btn:FindFirstChild("Label").TextColor3=Theme.TabInactiveText
                prev._btn:FindFirstChild("Label").Font=Enum.Font.Gotham
                prev._scroll.Visible=false
            end
            self._activeTab=Tab
            Tween(tabBtn,{BackgroundColor3=Theme.TabActiveBg,BackgroundTransparency=0.88},TI_MID)
            Tween(accentBar,{BackgroundTransparency=0},TI_MID)
            iconLbl.TextColor3=Theme.TabActiveText
            nameLbl.TextColor3=Theme.TabActiveText
            nameLbl.Font=Enum.Font.GothamSemibold
            tabScroll.Visible=true
        end

        tabBtn.MouseButton1Click:Connect(Activate)
        tabBtn.MouseEnter:Connect(function()
            if self._activeTab~=Tab then
                Tween(tabBtn,{BackgroundColor3=Theme.TabHover,BackgroundTransparency=0.5},TI_FAST)
            end
        end)
        tabBtn.MouseLeave:Connect(function()
            if self._activeTab~=Tab then
                Tween(tabBtn,{BackgroundTransparency=1},TI_FAST)
            end
        end)

        if #self._tabs==0 then task.defer(Activate) end
        table.insert(self._tabs, Tab)

        -- ── SECTION HEADER ───────────────────────────────────────────────
        function Tab:AddSection(sc)
            sc=sc or {}
            self._order=self._order+1
            local hasDesc = sc.Description and sc.Description~=""
            local h = hasDesc and SECTION_H or SECTION_H_S

            local frame = Create("Frame", {
                Name="Section", Size=UDim2.new(1,0,0,h),
                BackgroundTransparency=1, LayoutOrder=self._order, Parent=self._content,
            })

            -- Left accent stripe
            local stripe = Create("Frame", {
                Name="Stripe", Size=UDim2.new(0,3,0,hasDesc and 32 or 22),
                Position=UDim2.new(0,16,0.5,0), AnchorPoint=Vector2.new(0,0.5),
                BackgroundColor3=Theme.SectionAccent, BorderSizePixel=0, Parent=frame,
            })
            Round(stripe,2)

            Create("TextLabel", {
                Size=UDim2.new(1,-40,0,20), Position=UDim2.new(0,26,0,hasDesc and 10 or 11),
                BackgroundTransparency=1, Text=sc.Title or "",
                TextColor3=Theme.TitleText, TextSize=14, Font=Enum.Font.GothamBold,
                TextXAlignment=Enum.TextXAlignment.Left, Parent=frame,
            })
            if hasDesc then
                Create("TextLabel", {
                    Size=UDim2.new(1,-40,0,14), Position=UDim2.new(0,26,0,30),
                    BackgroundTransparency=1, Text=sc.Description,
                    TextColor3=Theme.DescText, TextSize=11, Font=Enum.Font.Gotham,
                    TextXAlignment=Enum.TextXAlignment.Left, Parent=frame,
                })
            end

            -- Bottom separator
            Create("Frame", {
                Size=UDim2.new(1,-24,0,1), Position=UDim2.new(0,12,1,-1),
                BackgroundColor3=Theme.Separator, BorderSizePixel=0, Parent=frame,
            })
        end

        -- ── ROW HELPERS ──────────────────────────────────────────────────
        local function MakeRow(tab, h)
            tab._order=tab._order+1
            local hasH = h~=nil
            local row = Create("Frame", {
                Name="Row_"..tab._order,
                Size=UDim2.new(1,0,0,h or ROW_H),
                BackgroundTransparency=1,
                LayoutOrder=tab._order, Parent=tab._content,
            })
            -- Hover background
            local hbg = Create("Frame", {
                Name="HoverBg", Size=UDim2.new(1,0,1,0),
                BackgroundColor3=Theme.RowHover, BackgroundTransparency=1,
                BorderSizePixel=0, ZIndex=row.ZIndex, Parent=row,
            })
            row.MouseEnter:Connect(function()
                Tween(hbg,{BackgroundTransparency=0.75},TI_FAST)
            end)
            row.MouseLeave:Connect(function()
                Tween(hbg,{BackgroundTransparency=1},TI_FAST)
            end)
            -- Thin separator
            Create("Frame", {
                Name="Sep", Size=UDim2.new(1,-24,0,1), Position=UDim2.new(0,12,1,-1),
                BackgroundColor3=Theme.Separator, BorderSizePixel=0, Parent=row,
            })
            return row
        end

        local function RowLabels(row, name, desc)
            local hasDesc = desc and desc~=""
            Create("TextLabel", {
                Name="Name",
                Size=UDim2.new(0.52,-20,0,18),
                Position=UDim2.new(0,20,0,hasDesc and 13 or 0),
                AnchorPoint=Vector2.new(0, hasDesc and 0 or 0),
                BackgroundTransparency=1, Text=name or "",
                TextColor3=Theme.LabelText, TextSize=13, Font=Enum.Font.GothamSemibold,
                TextXAlignment=Enum.TextXAlignment.Left,
                TextTruncate=Enum.TextTruncate.AtEnd, Parent=row,
            })
            if hasDesc then
                Create("TextLabel", {
                    Name="Desc",
                    Size=UDim2.new(0.65,-20,0,14),
                    Position=UDim2.new(0,20,0,32),
                    BackgroundTransparency=1, Text=desc,
                    TextColor3=Theme.DescText, TextSize=11, Font=Enum.Font.Gotham,
                    TextXAlignment=Enum.TextXAlignment.Left,
                    TextTruncate=Enum.TextTruncate.AtEnd, Parent=row,
                })
            end
        end

        -- ── LABEL ────────────────────────────────────────────────────────
        function Tab:AddLabel(lc)
            lc=lc or {}
            local row=MakeRow(self)
            RowLabels(row, lc.Name, lc.Description)
            local vl=Create("TextLabel", {
                Name="Value",
                Size=UDim2.new(0.46,-20,1,0), Position=UDim2.new(0.54,0,0,0),
                BackgroundTransparency=1, Text=tostring(lc.Value or ""),
                TextColor3=Theme.ValueText, TextSize=13, Font=Enum.Font.Gotham,
                TextXAlignment=Enum.TextXAlignment.Right,
                TextTruncate=Enum.TextTruncate.AtEnd, Parent=row,
            })
            AddPadding(vl,0,0,0,20)
            local obj={}
            function obj:Set(v) vl.Text=tostring(v) end
            return obj
        end

        -- ── TOGGLE ───────────────────────────────────────────────────────
        function Tab:AddToggle(tc)
            tc=tc or {}
            local val      = tc.Default  or false
            local callback = tc.Callback or function() end
            local row=MakeRow(self)
            RowLabels(row, tc.Name, tc.Description)

            -- Track
            local track=Create("Frame", {
                Size=UDim2.new(0,46,0,26), Position=UDim2.new(1,-64,0.5,-13),
                BackgroundColor3=val and Theme.ToggleOn or Theme.ToggleOff,
                BorderSizePixel=0, Parent=row,
            })
            Round(track,13)

            -- Track inner shadow (subtle inset)
            Create("UIStroke", {
                Color=Color3.fromRGB(0,0,0), Thickness=1, Transparency=0.7, Parent=track,
            })

            -- Knob
            local knob=Create("Frame", {
                Size=UDim2.new(0,22,0,22),
                Position=val and UDim2.new(0,22,0.5,-11) or UDim2.new(0,2,0.5,-11),
                BackgroundColor3=Theme.ToggleKnob, BorderSizePixel=0, ZIndex=track.ZIndex+1,
                Parent=track,
            })
            Round(knob,11)
            -- Knob drop shadow
            Create("UIStroke", {Color=Color3.fromRGB(0,0,0),Thickness=1,Transparency=0.8,Parent=knob})

            local btn=Create("TextButton", {
                Size=UDim2.new(1,0,1,0), BackgroundTransparency=1,
                Text="", Parent=track,
            })

            local obj={}
            local function Apply(v, silent)
                val=v
                if val then
                    Tween(track,{BackgroundColor3=Theme.ToggleOn},TI_MID)
                    Tween(knob,{Position=UDim2.new(0,22,0.5,-11)},TI_MID)
                else
                    Tween(track,{BackgroundColor3=Theme.ToggleOff},TI_MID)
                    Tween(knob,{Position=UDim2.new(0,2,0.5,-11)},TI_MID)
                end
                Crispylib._flags[tc.Flag]=val
                if not silent then pcall(callback,val) end
            end

            btn.MouseButton1Click:Connect(function() Apply(not val) end)
            function obj:Set(v,s) Apply(v,s) end
            function obj:Get() return val end
            RegisterFlag(tc.Flag, function() return val end, function(v) Apply(v,true) end)
            return obj
        end

        -- ── DROPDOWN ─────────────────────────────────────────────────────
        function Tab:AddDropdown(dc)
            dc=dc or {}
            local options  = dc.Options  or {}
            local multi    = dc.Multi    or false
            local callback = dc.Callback or function() end
            local selected = dc.Default  or (multi and {} or (options[1] or ""))

            local row=MakeRow(self)
            RowLabels(row, dc.Name, dc.Description)

            local function ValText()
                if multi then
                    return (#selected>0) and table.concat(selected,", ") or "None"
                end
                return tostring(selected~="" and selected or "None")
            end

            -- Dropdown button
            local dBtn=Create("TextButton", {
                Name="DropBtn",
                Size=UDim2.new(0,168,0,30), Position=UDim2.new(1,-186,0.5,-15),
                BackgroundColor3=Theme.DropdownBg, Text="",
                BorderSizePixel=0, AutoButtonColor=false, Parent=row,
            })
            Round(dBtn,8)
            local dStroke=Stroke(dBtn, Theme.DropdownBorder, 1)

            local dLabel=Create("TextLabel", {
                Name="Label",
                Size=UDim2.new(1,-32,1,0), Position=UDim2.new(0,10,0,0),
                BackgroundTransparency=1, Text=ValText(),
                TextColor3=Theme.ValueText, TextSize=12, Font=Enum.Font.Gotham,
                TextXAlignment=Enum.TextXAlignment.Left,
                TextTruncate=Enum.TextTruncate.AtEnd, Parent=dBtn,
            })
            -- Chevron icon
            local chevron=Create("TextLabel", {
                Name="Chevron",
                Size=UDim2.new(0,22,1,0), Position=UDim2.new(1,-24,0,0),
                BackgroundTransparency=1, Text="⌄",
                TextColor3=Theme.SubtitleText, TextSize=16, Font=Enum.Font.GothamBold,
                Parent=dBtn,
            })

            local listOpen=false
            local listFrame=Create("Frame", {
                Name="DropList",
                Size=UDim2.new(0,188,0,0), Position=UDim2.new(1,-188,1,6),
                BackgroundColor3=Theme.DropdownBg, BorderSizePixel=0,
                ZIndex=30, ClipsDescendants=true, Visible=false, Parent=row,
            })
            Round(listFrame,9)
            Stroke(listFrame, Theme.DropdownBorder, 1)

            -- Search inside dropdown
            local searchF=Create("Frame", {
                Size=UDim2.new(1,-16,0,28), Position=UDim2.new(0,8,0,6),
                BackgroundColor3=Theme.InputBg, BorderSizePixel=0, ZIndex=31, Parent=listFrame,
            })
            Round(searchF,7)
            local sStk=Stroke(searchF, Theme.DropdownBorder, 1)
            local sIn=Create("TextBox", {
                Size=UDim2.new(1,-10,1,0), Position=UDim2.new(0,8,0,0),
                BackgroundTransparency=1, PlaceholderText="Search...",
                PlaceholderColor3=Theme.SearchPlaceholder, Text="",
                TextColor3=Theme.LabelText, TextSize=12, Font=Enum.Font.Gotham,
                TextXAlignment=Enum.TextXAlignment.Left, ClearTextOnFocus=false,
                ZIndex=32, Parent=searchF,
            })
            sIn.Focused:Connect(function() Tween(sStk,{Color=Theme.FocusBorder},TI_FAST) end)
            sIn.FocusLost:Connect(function() Tween(sStk,{Color=Theme.DropdownBorder},TI_FAST) end)

            local optScroll=Create("ScrollingFrame", {
                Size=UDim2.new(1,-8,1,-46), Position=UDim2.new(0,4,0,40),
                BackgroundTransparency=1, ScrollBarThickness=3,
                ScrollBarImageColor3=Theme.ScrollbarThumb, BorderSizePixel=0,
                CanvasSize=UDim2.new(0,0,0,0), AutomaticCanvasSize=Enum.AutomaticSize.Y,
                ZIndex=31, Parent=listFrame,
            })
            local optList=Create("Frame", {
                Size=UDim2.new(1,0,0,0), AutomaticSize=Enum.AutomaticSize.Y,
                BackgroundTransparency=1, ZIndex=31, Parent=optScroll,
            })
            ListLayout(optList, Enum.FillDirection.Vertical, 2)
            AddPadding(optList,2,2,0,0)

            local function IsSel(opt)
                if multi then
                    for _,v in ipairs(selected) do if v==opt then return true end end
                    return false
                end
                return selected==opt
            end

            local optBtns={}
            local function Rebuild(filter)
                for _,c in ipairs(optBtns) do c:Destroy() end
                optBtns={}
                filter=(filter or ""):lower()
                for _,opt in ipairs(options) do
                    if filter=="" or opt:lower():find(filter,1,true) then
                        local sel=IsSel(opt)
                        local ob=Create("TextButton", {
                            Size=UDim2.new(1,0,0,30),
                            BackgroundColor3=sel and Theme.Accent or Theme.DropdownBg,
                            BackgroundTransparency=sel and 0.85 or 1,
                            Text="", BorderSizePixel=0, AutoButtonColor=false,
                            ZIndex=32, Parent=optList,
                        })
                        Round(ob,6)
                        Create("TextLabel", {
                            Size=UDim2.new(1,-16,1,0), Position=UDim2.new(0,10,0,0),
                            BackgroundTransparency=1, Text=opt,
                            TextColor3=sel and Theme.TabActiveText or Theme.LabelText,
                            TextSize=12, Font=sel and Enum.Font.GothamSemibold or Enum.Font.Gotham,
                            TextXAlignment=Enum.TextXAlignment.Left,
                            TextTruncate=Enum.TextTruncate.AtEnd, ZIndex=33, Parent=ob,
                        })
                        -- Checkmark for multi
                        if multi and sel then
                            Create("TextLabel", {
                                Size=UDim2.new(0,18,1,0), Position=UDim2.new(1,-20,0,0),
                                BackgroundTransparency=1, Text="✓",
                                TextColor3=Theme.Accent, TextSize=12, Font=Enum.Font.GothamBold,
                                ZIndex=33, Parent=ob,
                            })
                        end
                        ob.MouseEnter:Connect(function()
                            if not IsSel(opt) then
                                Tween(ob,{BackgroundColor3=Theme.DropdownItemHover,BackgroundTransparency=0},TI_FAST)
                            end
                        end)
                        ob.MouseLeave:Connect(function()
                            if not IsSel(opt) then Tween(ob,{BackgroundTransparency=1},TI_FAST) end
                        end)
                        ob.MouseButton1Click:Connect(function()
                            if multi then
                                local found=false
                                for i,v in ipairs(selected) do
                                    if v==opt then table.remove(selected,i); found=true; break end
                                end
                                if not found then table.insert(selected,opt) end
                            else
                                selected=opt
                                listOpen=false
                                Tween(listFrame,{Size=UDim2.new(0,188,0,0)},TI_MID)
                                Tween(chevron,{Rotation=0},TI_MID)
                                Tween(dStroke,{Color=Theme.DropdownBorder},TI_MID)
                                task.delay(0.22,function() listFrame.Visible=false end)
                            end
                            dLabel.Text=ValText()
                            Crispylib._flags[dc.Flag]=selected
                            pcall(callback,selected)
                            Rebuild(sIn.Text)
                        end)
                        table.insert(optBtns,ob)
                    end
                end
            end

            Rebuild()
            sIn:GetPropertyChangedSignal("Text"):Connect(function() Rebuild(sIn.Text) end)

            local MAX_H=math.min(#options*32+50,230)
            dBtn.MouseButton1Click:Connect(function()
                listOpen=not listOpen
                if listOpen then
                    listFrame.Visible=true; sIn.Text=""; Rebuild()
                    Tween(listFrame,{Size=UDim2.new(0,188,0,MAX_H)},TI_MID)
                    Tween(chevron,{Rotation=180},TI_MID)
                    Tween(dStroke,{Color=Theme.FocusBorder},TI_MID)
                else
                    Tween(listFrame,{Size=UDim2.new(0,188,0,0)},TI_MID)
                    Tween(chevron,{Rotation=0},TI_MID)
                    Tween(dStroke,{Color=Theme.DropdownBorder},TI_MID)
                    task.delay(0.22,function() listFrame.Visible=false end)
                end
            end)

            local obj={}
            function obj:Set(v,s)
                selected=v; dLabel.Text=ValText()
                Crispylib._flags[dc.Flag]=selected
                if not s then pcall(callback,selected) end; Rebuild()
            end
            function obj:Get() return selected end
            function obj:SetOptions(newOpts) options=newOpts; Rebuild() end
            RegisterFlag(dc.Flag, function() return selected end, function(v) obj:Set(v,true) end)
            return obj
        end

        -- ── SLIDER ───────────────────────────────────────────────────────
        function Tab:AddSlider(sc)
            sc=sc or {}
            local min      = sc.Min      or 0
            local max      = sc.Max      or 100
            local suffix   = sc.Suffix   or ""
            local callback = sc.Callback or function() end
            local val      = math.clamp(sc.Default or min, min, max)
            local step     = sc.Step     or 1

            local row=MakeRow(self, 70)
            RowLabels(row, sc.Name, sc.Description)

            -- Value display
            local vLbl=Create("TextLabel", {
                Name="Val", Size=UDim2.new(0,70,0,18), Position=UDim2.new(1,-88,0,13),
                BackgroundTransparency=1, Text=tostring(val)..suffix,
                TextColor3=Theme.Accent, TextSize=13, Font=Enum.Font.GothamBold,
                TextXAlignment=Enum.TextXAlignment.Right, Parent=row,
            })

            -- Track background
            local trackBg=Create("Frame", {
                Size=UDim2.new(1,-40,0,6), Position=UDim2.new(0,20,1,-20),
                BackgroundColor3=Theme.ToggleOff, BorderSizePixel=0, Parent=row,
            })
            Round(trackBg,4)

            -- Fill
            local fill=Create("Frame", {
                Size=UDim2.new((val-min)/(max-min),0,1,0),
                BackgroundColor3=Theme.Accent, BorderSizePixel=0, Parent=trackBg,
            })
            Round(fill,4)

            -- Knob
            local knob=Create("Frame", {
                Size=UDim2.new(0,16,0,16), AnchorPoint=Vector2.new(0.5,0.5),
                Position=UDim2.new((val-min)/(max-min),0,0.5,0),
                BackgroundColor3=Color3.fromRGB(255,255,255), BorderSizePixel=0,
                ZIndex=trackBg.ZIndex+2, Parent=trackBg,
            })
            Round(knob,8)
            Create("UIStroke",{Color=Theme.Accent,Thickness=2,Parent=knob})

            -- Invisible wider hit area
            local hitArea=Create("TextButton", {
                Size=UDim2.new(1,0,0,28), Position=UDim2.new(0,0,0.5,-14),
                BackgroundTransparency=1, Text="", BorderSizePixel=0,
                ZIndex=trackBg.ZIndex+3, Parent=trackBg,
            })

            local dragging=false
            local function UpdateSlider(pos)
                local abs=trackBg.AbsolutePosition; local sz=trackBg.AbsoluteSize
                local t=math.clamp((pos.X-abs.X)/sz.X,0,1)
                val=math.floor(min+t*(max-min)/step+0.5)*step
                val=math.clamp(val,min,max)
                local frac=(val-min)/(max-min)
                fill.Size=UDim2.new(frac,0,1,0)
                knob.Position=UDim2.new(frac,0,0.5,0)
                vLbl.Text=tostring(val)..suffix
                Crispylib._flags[sc.Flag]=val
                pcall(callback,val)
            end

            hitArea.InputBegan:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 then
                    dragging=true; UpdateSlider(i.Position)
                    Tween(knob,{Size=UDim2.new(0,18,0,18)},TI_FAST)
                end
            end)
            UserInputService.InputChanged:Connect(function(i)
                if dragging and i.UserInputType==Enum.UserInputType.MouseMovement then
                    UpdateSlider(i.Position)
                end
            end)
            UserInputService.InputEnded:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 then
                    dragging=false
                    Tween(knob,{Size=UDim2.new(0,16,0,16)},TI_FAST)
                end
            end)

            local obj={}
            function obj:Set(v,s)
                val=math.clamp(v,min,max)
                local frac=(val-min)/(max-min)
                fill.Size=UDim2.new(frac,0,1,0)
                knob.Position=UDim2.new(frac,0,0.5,0)
                vLbl.Text=tostring(val)..suffix
                Crispylib._flags[sc.Flag]=val
                if not s then pcall(callback,val) end
            end
            function obj:Get() return val end
            RegisterFlag(sc.Flag, function() return val end, function(v) obj:Set(v,true) end)
            return obj
        end

        -- ── KEYBIND ──────────────────────────────────────────────────────
        function Tab:AddKeybind(kc)
            kc=kc or {}
            local key      = kc.Default  or Enum.KeyCode.Unknown
            local callback = kc.Callback or function() end
            local listening=false
            local row=MakeRow(self)
            RowLabels(row, kc.Name, kc.Description)

            local kbBg=Create("Frame", {
                Size=UDim2.new(0,106,0,30), Position=UDim2.new(1,-124,0.5,-15),
                BackgroundColor3=Theme.InputBg, BorderSizePixel=0, Parent=row,
            })
            Round(kbBg,8); local kbStroke=Stroke(kbBg,Theme.InputBorder,1)

            local kbBtn=Create("TextButton", {
                Size=UDim2.new(1,0,1,0), BackgroundTransparency=1,
                Text=key.Name, TextColor3=Theme.ValueText,
                TextSize=12, Font=Enum.Font.GothamSemibold, BorderSizePixel=0, Parent=kbBg,
            })

            kbBtn.MouseButton1Click:Connect(function()
                if listening then return end
                listening=true; kbBtn.Text="Press..."; kbBtn.TextColor3=Theme.Accent
                Tween(kbStroke,{Color=Theme.FocusBorder},TI_FAST)
            end)
            UserInputService.InputBegan:Connect(function(i)
                if not listening then return end
                if i.UserInputType~=Enum.UserInputType.Keyboard then return end
                key=i.KeyCode; listening=false
                kbBtn.Text=key.Name; kbBtn.TextColor3=Theme.ValueText
                Tween(kbStroke,{Color=Theme.InputBorder},TI_FAST)
            end)
            UserInputService.InputBegan:Connect(function(i,gpe)
                if gpe then return end
                if not listening and i.UserInputType==Enum.UserInputType.Keyboard and i.KeyCode==key then
                    pcall(callback,key)
                end
            end)

            local obj={}
            function obj:Set(k) key=k; kbBtn.Text=k.Name end
            function obj:Get() return key end
            return obj
        end

            cpc=cpc or {}

            Round(sbKnob,6); Create("UIStroke",{Color=Color3.fromRGB(0,0,0),Transparency=0.6,Thickness=1,Parent=sbKnob})

                Size=UDim2.new(1,-16,0,14), Position=UDim2.new(0,8,0,146),
                Image="rbxassetid://6923790674", ZIndex=31, Parent=pFrame,
            })
            Round(hBar,4)

            local hKnob=Create("Frame", {
                Size=UDim2.new(0,10,1,4), AnchorPoint=Vector2.new(0.5,0.5),
                Position=UDim2.new(0,0,0.5,0), BackgroundColor3=Color3.fromRGB(255,255,255),
                BorderSizePixel=0, ZIndex=32, Parent=hBar,
            })
            Round(hKnob,3); Create("UIStroke",{Color=Color3.fromRGB(80,80,80),Thickness=1,Parent=hKnob})

            local cPrev=Create("Frame", {
                Size=UDim2.new(1,-16,0,22), Position=UDim2.new(0,8,0,168),
                BackgroundColor3=val, BorderSizePixel=0, ZIndex=31, Parent=pFrame,
            })
            Round(cPrev,6)

            local h,s,br=Color3.toHSV(val)
            local function UpdateColor()
                val=Color3.fromHSV(h,s,br)
                swatch.BackgroundColor3=val; cPrev.BackgroundColor3=val
                sbBox.BackgroundColor3=Color3.fromHSV(h,1,1)
                sbKnob.Position=UDim2.new(s,-6,1-br,-6)
                hKnob.Position=UDim2.new(h,-5,0.5,0)
                Crispylib._flags[cpc.Flag]=val; pcall(callback,val)
            end
            UpdateColor()

                local a=sbBox.AbsolutePosition; local z=sbBox.AbsoluteSize
                s=math.clamp((pos.X-a.X)/z.X,0,1)
                br=math.clamp(1-(pos.Y-a.Y)/z.Y,0,1); UpdateColor()

                    if dSB then SBDrag(i.Position) end
                    if dH  then HDrag(i.Position)  end
                end

            UserInputService.InputEnded:Connect(function(i)
                if i.UserInputType==Enum.UserInputType.MouseButton1 then dSB=false; dH=false end

                val=color; h,s2,br2=Color3.toHSV(color); h=h; s=s2; br=br2
                UpdateColor(); if not s then pcall(callback,val) end
            end
            function obj:Get() return val end
            RegisterFlag(cpc.Flag, function() return val end, function(v) obj:Set(v,true) end)
            return obj
        end
