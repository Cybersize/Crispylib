local function Create(class, props, children)
    local obj = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then obj[k] = v end
    end
    for _, child in ipairs(children or {}) do child.Parent = obj end
    if props and props.Parent then obj.Parent = props.Parent end
    return obj
end

local function Tween(obj, goals, info)
    TweenService:Create(obj, info or TWEEN_INFO, goals):Play()
end

local function MakeDraggable(titleBar, frame)
    local dragging, dragStart, startPos
    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging  = true
            dragStart = input.Position
            startPos  = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
end

-- ────────────────────────────────────────────────────────────────────────────
--  CONFIG / SERIALISATION SYSTEM
-- ────────────────────────────────────────────────────────────────────────────
--  Each flagged element registers itself in Crispylib._flagElements[flag]
--  as an object with :Get() and :Set(v). The Config module reads/writes these.
--
--  Storage priority:
--    1. writefile / readfile  (executor environment)
--    2. In-memory only (prints a warning; DataStore is server-only)

local Crispylib = {}
Crispylib.__index = Crispylib
Crispylib.Theme        = Theme
Crispylib._flags       = {}        -- flag -> current value (mirrors element state)
Crispylib._flagElements = {}       -- flag -> { Get, Set }  element handle
Crispylib._configName  = "Crispylib"

-- Detect executor file API
local _hasWritefile = (type(writefile) == "function")
local _hasReadfile  = (type(readfile)  == "function")
local _hasListfiles = (type(listfiles) == "function")
local _hasIsfile    = (type(isfile)    == "function")
local _hasMakefolder = (type(makefolder) == "function")

-- ── Serialisation helpers ───────────────────────────────────────────────────
local function SerialiseValue(v)
    local t = typeof(v)
    if t == "Color3" then
        return { __type = "Color3", r = v.R, g = v.G, b = v.B }
    elseif t == "table" then
        local out = {}
        for i, item in ipairs(v) do out[i] = SerialiseValue(item) end
        return out
    else
        return v
    end
end

local function DeserialiseValue(v)
    if type(v) == "table" then
        if v.__type == "Color3" then
            return Color3.new(v.r, v.g, v.b)
        else
            local out = {}
            for i, item in ipairs(v) do out[i] = DeserialiseValue(item) end
            return out
        end
    end
    return v
end

-- ── File paths ──────────────────────────────────────────────────────────────
local function ConfigFolder()
    return "Crispylib/" .. Crispylib._configName
end

local function ProfilePath(name)
    return ConfigFolder() .. "/" .. tostring(name) .. ".json"
end

local function EnsureFolder()
    if _hasMakefolder then
        pcall(makefolder, "Crispylib")
        pcall(makefolder, ConfigFolder())
    end
end

-- ── Config public API ───────────────────────────────────────────────────────
Crispylib.Config = {}

--- Snapshot all flagged values into a flat table
function Crispylib.Config.Snapshot()
    local snap = {}
    for flag, elem in pairs(Crispylib._flagElements) do
        snap[flag] = SerialiseValue(elem.Get())
    end
    return snap
end

--- Apply a snapshot table to all matching flagged elements
function Crispylib.Config.Apply(snap)
    for flag, raw in pairs(snap) do
        local elem = Crispylib._flagElements[flag]
        if elem then
            pcall(elem.Set, DeserialiseValue(raw))
        end
    end
end

--- Export current config as a JSON string
function Crispylib.Config.Export()
    local ok, json = pcall(function()
        return HttpService:JSONEncode(Crispylib.Config.Snapshot())
    end)
    return ok and json or "{}"
end

--- Import and apply config from a JSON string
function Crispylib.Config.Import(json)
    local ok, snap = pcall(function()
        return HttpService:JSONDecode(json)
    end)
    if ok and type(snap) == "table" then
        Crispylib.Config.Apply(snap)
        return true
    end
    return false
end

--- Save current config to a named profile file
function Crispylib.Config.Save(name)
    name = name or "default"
    if not _hasWritefile then
        warn("[Crispylib] writefile not available — config saved in-memory only")
        Crispylib._inMemory = Crispylib._inMemory or {}
        Crispylib._inMemory[name] = Crispylib.Config.Snapshot()
        return true
    end
    EnsureFolder()
    local ok = pcall(writefile, ProfilePath(name), Crispylib.Config.Export())
    return ok
end

--- Load a named profile and apply it
function Crispylib.Config.Load(name)
    name = name or "default"
    if not _hasReadfile then
        -- Try in-memory fallback
        if Crispylib._inMemory and Crispylib._inMemory[name] then
            Crispylib.Config.Apply(Crispylib._inMemory[name])
            return true
        end
        return false
    end
    local path = ProfilePath(name)
    local exists = true
    if _hasIsfile then exists = isfile(path) end
    if not exists then return false end
    local ok, json = pcall(readfile, path)
    if not ok then return false end
    return Crispylib.Config.Import(json)
end

--- List all saved profile names
function Crispylib.Config.List()
    if not _hasListfiles then
        -- Return in-memory keys
        local list = {}
        if Crispylib._inMemory then
            for k in pairs(Crispylib._inMemory) do list[#list+1] = k end
        end
        return list
    end
    EnsureFolder()
    local ok, files = pcall(listfiles, ConfigFolder())
    if not ok then return {} end
    local names = {}
    for _, path in ipairs(files) do
        local name = path:match("([^/\\]+)%.json$")
        if name then names[#names+1] = name end
    end
    return names
end

--- Delete a saved profile
function Crispylib.Config.Delete(name)
    if not name then return false end
    if not _hasWritefile then
        if Crispylib._inMemory then Crispylib._inMemory[name] = nil end
        return true
    end
    local path = ProfilePath(name)
    if _hasIsfile and not isfile(path) then return false end
    local ok = pcall(writefile, path, "__deleted__")
    -- Some executors have delfile; try it
    if type(delfile) == "function" then
        pcall(delfile, path)
    else
        -- overwrite with empty to at least clear it
        pcall(writefile, path, "{}")
    end
    return ok
end

-- ── Internal flag registration ───────────────────────────────────────────────
local function RegisterFlag(flag, getFunc, setFunc)
    if not flag then return end
    Crispylib._flagElements[flag] = { Get = getFunc, Set = setFunc }
end

--- Show a notification toast.
-- @param config  { Title, Description, Duration, Type }
-- Type: "info" | "success" | "error" | "warn"
function Crispylib.Notify(config)
    config = config or {}
    EnsureNotifGui()

    local typeColor = ({
        info    = Theme.NotifInfo,
        success = Theme.NotifSuccess,
        error   = Theme.NotifError,
        warn    = Theme.NotifWarn,
    })[config.Type or "info"] or Theme.NotifInfo

    local duration = config.Duration or 4

    local notif = Create("Frame", {
        Name             = "Notif",
        Size             = UDim2.new(1, 0, 0, 70),
        BackgroundColor3 = Theme.NotifBg,
        BackgroundTransparency = 0,
        BorderSizePixel  = 0,
        Parent           = _notifContainer,
    })
    Round(notif, 10)
    Stroke(notif, Theme.NotifBorder, 1)

    -- Coloured left accent
    local accent = Create("Frame", {
        Name             = "Accent",
        Size             = UDim2.new(0, 4, 1, -16),
        Position         = UDim2.new(0, 8, 0.5, 0),
        AnchorPoint      = Vector2.new(0, 0.5),
        BackgroundColor3 = typeColor,
        BorderSizePixel  = 0,
        Parent           = notif,
    })
    Round(accent, 2)

    Create("TextLabel", {
        Name             = "Title",
        Size             = UDim2.new(1, -28, 0, 20),
        Position         = UDim2.new(0, 22, 0, 12),
        BackgroundTransparency = 1,
        Text             = config.Title or "Notification",
        TextColor3       = Theme.TitleText,
        TextSize         = 13,
        Font             = Enum.Font.GothamBold,
        TextXAlignment   = Enum.TextXAlignment.Left,
        Parent           = notif,
    })
    Create("TextLabel", {
        Name             = "Desc",
        Size             = UDim2.new(1, -28, 0, 28),
        Position         = UDim2.new(0, 22, 0, 32),
        BackgroundTransparency = 1,
        Text             = config.Description or "",
        TextColor3       = Theme.DescText,
        TextSize         = 11,
        Font             = Enum.Font.Gotham,
        TextXAlignment   = Enum.TextXAlignment.Left,
        TextWrapped      = true,
        Parent           = notif,
    })

    -- Progress bar
    local progressBg = Create("Frame", {
        Name             = "ProgressBg",
        Size             = UDim2.new(1, -16, 0, 3),
        Position         = UDim2.new(0, 8, 1, -6),
        BackgroundColor3 = Theme.LoaderBarBg,
        BorderSizePixel  = 0,
        Parent           = notif,
    })
    Round(progressBg, 2)
    local progressFill = Create("Frame", {
        Name             = "Fill",
        Size             = UDim2.new(1, 0, 1, 0),
        BackgroundColor3 = typeColor,
        BorderSizePixel  = 0,
        Parent           = progressBg,

    notif.Position = UDim2.new(1, 20, 0, 0)

    -- Drain progress bar
    Tween(progressFill, { Size = UDim2.new(0, 0, 1, 0) },
        TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out))

    -- Dismiss after duration
    task.delay(duration, function()
        Tween(notif, { Position = UDim2.new(1, 20, 0, 0) }, TWEEN_EASE)
        task.delay(0.4, function()
            notif:Destroy()
        end)
    end)
end

--- Creates a full-screen loading overlay.
-- config: { Title, Subtitle, LogoId, MinimumTime }
-- Returns: { SetProgress(0-1), SetStatus(text), AddTask(text), Finish() }
function Crispylib.CreateLoadingScreen(config)
    config = config or {}
    local title    = config.Title    or "Loading"
    local subtitle = config.Subtitle or ""
    local logoId   = config.LogoId   or ""
    local minTime  = config.MinimumTime or 0

    local startTime = os.clock()

    local sg = Create("ScreenGui", {
        Name           = "Crispylib_Loader",
        ResetOnSpawn   = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        IgnoreGuiInset = true,
        DisplayOrder   = 2000,
    })
    pcall(function() sg.Parent = CoreGui end)
    if not sg.Parent then sg.Parent = LocalPlayer:WaitForChild("PlayerGui") end

    -- Full dark background
    local bg = Create("Frame", {
        Name             = "Bg",
        Size             = UDim2.new(1, 0, 1, 0),
        BackgroundColor3 = Theme.LoaderBg,
        BorderSizePixel  = 0,
        Parent           = sg,
    })

    -- Subtle animated gradient
    local grad = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(22, 22, 26)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(14, 14, 18)),
        }),
        Rotation = 135,
        Parent   = bg,
    })

    -- Center container
    local center = Create("Frame", {
        Name             = "Center",
        Size             = UDim2.new(0, 340, 0, 260),
        Position         = UDim2.new(0.5, -170, 0.5, -130),
        BackgroundTransparency = 1,
        Parent           = bg,
    })

    -- Logo image (if provided)
    local logoY = 0
    if logoId ~= "" then
        Create("ImageLabel", {
            Name             = "Logo",
            Size             = UDim2.new(0, 64, 0, 64),
            Position         = UDim2.new(0.5, -32, 0, 0),
            BackgroundTransparency = 1,
            Image            = logoId,
            Parent           = center,
        })
        logoY = 76
    end

    -- Title
    Create("TextLabel", {
        Name             = "Title",
        Size             = UDim2.new(1, 0, 0, 32),
        Position         = UDim2.new(0, 0, 0, logoY),
        BackgroundTransparency = 1,
        Text             = title,
        TextColor3       = Theme.TitleText,
        TextSize         = 24,
        Font             = Enum.Font.GothamBold,
        TextXAlignment   = Enum.TextXAlignment.Center,
        Parent           = center,
    })

    -- Subtitle
    local subtitleLabel = Create("TextLabel", {
        Name             = "Subtitle",
        Size             = UDim2.new(1, 0, 0, 20),
        Position         = UDim2.new(0, 0, 0, logoY + 36),
        BackgroundTransparency = 1,
        Text             = subtitle,
        TextColor3       = Theme.SubtitleText,
        TextSize         = 13,
        Font             = Enum.Font.Gotham,
        TextXAlignment   = Enum.TextXAlignment.Center,
        Parent           = center,
    })

    -- Progress bar background
    local barBg = Create("Frame", {
        Name             = "BarBg",
        Size             = UDim2.new(1, 0, 0, 5),
        Position         = UDim2.new(0, 0, 0, logoY + 84),
        BackgroundColor3 = Theme.LoaderBarBg,
        BorderSizePixel  = 0,
        Parent           = center,
    })
    Round(barBg, 3)

    local barFill = Create("Frame", {
        Name             = "Fill",
        Size             = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = Theme.LoaderBarFill,
        BorderSizePixel  = 0,
        Parent           = barBg,
    })
    Round(barFill, 3)

    -- Status text
    local statusLabel = Create("TextLabel", {
        Name             = "Status",
        Size             = UDim2.new(1, 0, 0, 16),
        Position         = UDim2.new(0, 0, 0, logoY + 97),
        BackgroundTransparency = 1,
        Text             = "",
        TextColor3       = Theme.DescText,
        TextSize         = 11,
        Font             = Enum.Font.Gotham,
        TextXAlignment   = Enum.TextXAlignment.Center,
        Parent           = center,
    })

    -- Task list (scrollable, shows recent tasks)
    local taskList = Create("Frame", {
        Name             = "Tasks",
        Size             = UDim2.new(1, 0, 0, 100),
        Position         = UDim2.new(0, 0, 0, logoY + 120),
        BackgroundTransparency = 1,
        ClipsDescendants = true,
        Parent           = center,
    })
    ListLayout(taskList, Enum.FillDirection.Vertical, 4)

    local progress = 0

    local Loader = {}

    function Loader:SetProgress(pct)
        progress = math.clamp(pct, 0, 1)
        Tween(barFill, { Size = UDim2.new(progress, 0, 1, 0) },
            TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out))
    end

    function Loader:SetStatus(text)
        statusLabel.Text = tostring(text or "")
    end

    function Loader:AddTask(text)
        local lbl = Create("TextLabel", {
            Name             = "Task",
            Size             = UDim2.new(1, 0, 0, 16),
            BackgroundTransparency = 1,
            Text             = "✓  " .. tostring(text),
            TextColor3       = Theme.DescText,
            TextSize         = 11,
            Font             = Enum.Font.Gotham,
            TextXAlignment   = Enum.TextXAlignment.Center,
            Parent           = taskList,
        })
        -- Fade older tasks
        local children = taskList:GetChildren()
        for i, c in ipairs(children) do
            if c:IsA("TextLabel") then
                local alpha = math.clamp((i / #children), 0.2, 1)
                c.TextTransparency = 1 - alpha * 0.7
            end
        end
    end

    function Loader:Finish(callback)
        local elapsed = os.clock() - startTime
        local remaining = math.max(0, minTime - elapsed)
        task.delay(remaining, function()
            -- Fill to 100%
            self:SetProgress(1)
            task.wait(0.3)
            -- Fade out
            Tween(bg, { BackgroundTransparency = 1 }, TWEEN_SLOW)
            for _, desc in ipairs(bg:GetDescendants()) do
                if desc:IsA("TextLabel") or desc:IsA("ImageLabel") or desc:IsA("Frame") then
                    pcall(function()
                        Tween(desc, { BackgroundTransparency = 1, TextTransparency = 1 }, TWEEN_SLOW)
                    end)
                end
            end
            task.delay(0.35, function()
                sg:Destroy()
                if callback then pcall(callback) end
            end)
        end)
    end

    -- Gentle pulse on bar fill (shimmer)
    task.spawn(function()
        while sg and sg.Parent do
            Tween(barFill, { BackgroundColor3 = Color3.fromRGB(40, 160, 255) },
                TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut))
            task.wait(0.9)
            Tween(barFill, { BackgroundColor3 = Theme.LoaderBarFill },
                TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut))
            task.wait(0.9)
        end
    end)

    return Loader
end

                        ob.MouseLeave:Connect(function()
                            if not IsSelected(opt) then Tween(ob, { BackgroundTransparency = 1 }, TWEEN_INFO) end
                        end)
                        ob.MouseButton1Click:Connect(function()
                            if multi then
                                local found = false
                                for i, v in ipairs(selected) do
                                    if v == opt then table.remove(selected, i); found = true; break end
                                end
                                if not found then table.insert(selected, opt) end
                            else
                                selected = opt
                                listOpen = false
                                Tween(listFrame, { Size = UDim2.new(0, 180, 0, 0) }, TWEEN_INFO)
                                task.delay(0.2, function() listFrame.Visible = false end)
                            end
                            dropLabel.Text = ValueText()
                            pcall(callback, selected)
                            Crispylib._flags[dc.Flag] = selected
                            RebuildOptions(searchInput.Text)
                        end)
                        table.insert(optionBtns, ob)
                    end
                end
            end

            RebuildOptions()
            searchInput:GetPropertyChangedSignal("Text"):Connect(function() RebuildOptions(searchInput.Text) end)

            local MAX_H = math.min(#options * 30 + 46, 220)
            dropBtn.MouseButton1Click:Connect(function()
                listOpen = not listOpen
                if listOpen then
                    listFrame.Visible = true; searchInput.Text = ""; RebuildOptions()
                    Tween(listFrame, { Size = UDim2.new(0, 180, 0, MAX_H) }, TWEEN_INFO)
                else
                    Tween(listFrame, { Size = UDim2.new(0, 180, 0, 0) }, TWEEN_INFO)
                    task.delay(0.2, function() listFrame.Visible = false end)
                end
            end)

            local obj = {}
            function obj:Set(v, silent)
                selected = v; dropLabel.Text = ValueText()
                if not silent then pcall(callback, selected) end
                Crispylib._flags[dc.Flag] = selected
                RebuildOptions()
            end
            function obj:Get() return selected end
            function obj:SetOptions(newOpts) options = newOpts; RebuildOptions() end

            RegisterFlag(dc.Flag,
                function() return selected end,
                function(v) obj:Set(v, true) end)
            return obj
        end

        -- ── SLIDER ────────────────────────────────────────────────────────
        function Tab:AddSlider(sc)
            sc = sc or {}
            local min      = sc.Min      or 0
            local max      = sc.Max      or 100
            local suffix   = sc.Suffix   or ""
            local callback = sc.Callback or function() end
            local value    = math.clamp(sc.Default or min, min, max)

            local row = MakeRow(self, 66)
            AddRowLabels(row, sc.Name, sc.Description)

            local valLabel = Create("TextLabel", {
                Name = "Val", Size = UDim2.new(0, 80, 0, 18), Position = UDim2.new(1, -96, 0, 12),
                BackgroundTransparency = 1, Text = tostring(value)..suffix,
                TextColor3 = Theme.ValueText, TextSize = 12, Font = Enum.Font.Gotham,
                TextXAlignment = Enum.TextXAlignment.Right, Parent = row,
            })
            local trackBg = Create("Frame", {
                Size = UDim2.new(1, -32, 0, 5), Position = UDim2.new(0, 16, 1, -18),
                BackgroundColor3 = Theme.ToggleOff, BorderSizePixel = 0, Parent = row,
            })
            Round(trackBg, 3)
            local trackFill = Create("Frame", {
                Size = UDim2.new((value-min)/(max-min), 0, 1, 0),
                BackgroundColor3 = Theme.TabActive, BorderSizePixel = 0, Parent = trackBg,
            })
            Round(trackFill, 3)
            local knob = Create("Frame", {
                Size = UDim2.new(0, 14, 0, 14), AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new((value-min)/(max-min), 0, 0.5, 0),
                BackgroundColor3 = Theme.ToggleKnob, BorderSizePixel = 0, ZIndex = 5, Parent = trackBg,
            })
            Round(knob, 7)

            local dragging = false
            local function UpdateSlider(pos)
                local abs = trackBg.AbsolutePosition
                local sz  = trackBg.AbsoluteSize
                local t   = math.clamp((pos.X - abs.X) / sz.X, 0, 1)
                value = math.floor(min + t * (max - min) + 0.5)
                local frac = (value - min) / (max - min)
                trackFill.Size = UDim2.new(frac, 0, 1, 0)
                knob.Position  = UDim2.new(frac, 0, 0.5, 0)
                valLabel.Text  = tostring(value) .. suffix
                pcall(callback, value)
                Crispylib._flags[sc.Flag] = value
            end
            trackBg.InputBegan:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 then
                    dragging = true; UpdateSlider(i.Position)
                end
            end)
            UserInputService.InputChanged:Connect(function(i)
                if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then UpdateSlider(i.Position) end
            end)
            UserInputService.InputEnded:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
            end)

            local obj = {}
            function obj:Set(v, silent)
                value = math.clamp(v, min, max)
                local frac = (value-min)/(max-min)
                trackFill.Size = UDim2.new(frac, 0, 1, 0)
                knob.Position  = UDim2.new(frac, 0, 0.5, 0)
                valLabel.Text  = tostring(value)..suffix
                Crispylib._flags[sc.Flag] = value
                if not silent then pcall(callback, value) end
            end
            function obj:Get() return value end

            RegisterFlag(sc.Flag,
                function() return value end,
                function(v) obj:Set(v, true) end)
            return obj
        end

        -- ── KEYBIND ───────────────────────────────────────────────────────
        function Tab:AddKeybind(kc)
            kc = kc or {}
            local currentKey = kc.Default  or Enum.KeyCode.Unknown
            local callback   = kc.Callback or function() end
            local listening  = false
            local row = MakeRow(self)
            AddRowLabels(row, kc.Name, kc.Description)
            local kbBtn = Create("TextButton", {
                Size = UDim2.new(0, 100, 0, 28), Position = UDim2.new(1, -116, 0.5, -14),
                BackgroundColor3 = Theme.DropdownBg, Text = currentKey.Name,
                TextColor3 = Theme.ValueText, TextSize = 12, Font = Enum.Font.GothamSemibold,
                BorderSizePixel = 0, AutoButtonColor = false, Parent = row,
            })
            Round(kbBtn, 7); Stroke(kbBtn, Theme.DropdownBorder, 1)
            kbBtn.MouseButton1Click:Connect(function()
                if listening then return end
                listening = true; kbBtn.Text = "..."; kbBtn.TextColor3 = Theme.SubtitleText
            end)
            UserInputService.InputBegan:Connect(function(inp)
                if not listening then return end
                if inp.UserInputType ~= Enum.UserInputType.Keyboard then return end
                currentKey = inp.KeyCode; listening = false
                kbBtn.Text = currentKey.Name; kbBtn.TextColor3 = Theme.ValueText
            end)
            UserInputService.InputBegan:Connect(function(inp, gpe)
                if gpe then return end
                if not listening and inp.UserInputType == Enum.UserInputType.Keyboard and inp.KeyCode == currentKey then
                    pcall(callback, currentKey)
                end
            end)
            local obj = {}
            function obj:Set(key) currentKey = key; kbBtn.Text = key.Name end
            function obj:Get() return currentKey end
            return obj
        end

        -- ── COLOR PICKER ──────────────────────────────────────────────────

                Image = "rbxassetid://6923790674", ZIndex = 21, Parent = pickerFrame,

                satBrKnob.Position = UDim2.new(s, -6, 1 - br, -6)

                pcall(callback, value)

                if i.UserInputType == Enum.UserInputType.MouseButton1 then

                    h = math.clamp((i.Position.X - abs.X)/sz.X, 0, 1); UpdateColor()

                    Tween(pickerFrame, { Size = UDim2.new(0, 200, 0, 197) }, TWEEN_INFO)

                    task.delay(0.2, function() pickerFrame.Visible = false end)
                end
            end)

            function obj:Set(color, silent)
                value = color; h, s, br = Color3.toHSV(color); UpdateColor()
