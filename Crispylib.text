--[[
    CrispyLib v3  –  Lua UI Library for Roblox executors
    macOS-inspired dark GUI  –  Performance & Feature upgrade

    NEW in v3 (over v2):
    ─ Dropdown: AddOption(opt), RemoveOption(opt), SetOptions(list), GetOptions()
    ─ Dropdown: GetSelected() alias, proper multi-select state reset
    ─ Dropdown forward-reference bug fixed (Rebuild/CloseList ordering)
    ─ Sidebar resize: drag the divider to resize sidebar width (like MacLib)
    ─ Tab:SetVisible(bool) – show/hide entire tabs programmatically
    ─ Win:UpdateTitle(str), Win:UpdateSubtitle(str) – live title/subtitle update
    ─ UpdateName now actually works on Toggle, Slider, Input, Button, Keybind
    ─ Win:GetTab(name) – look up a tab object by name
    ─ CrispyLib.GetFlag(f) – shortcut to read any flag value
    ─ Connection pool: all connections tracked, properly disconnected on Unload
    ─ Drag state reuse: one shared InputChanged for window drag + slider + color
    ─ Notif duplicate guard: same Title+Type within 0.5s is silently ignored
    ─ Keybind listener leak fixed: bind event created once, not on every click
    ─ Slider: knob drag uses AbsoluteSize cached at drag-start, not every frame
    ─ ColorPicker: outside-click guard uses one shared connection, not per-open
    ─ Tab search: filter cleared when searchBox loses focus (optional opt-out)
    ─ ScrollingFrame scroll speed tuned (ScrollingFrame.ElasticBehavior = Never)
    ─ AddButton: optional Style = "Danger" renders red button
    ─ Separator now accepts optional Label text (renders inline label)
    ─ AddDropdown: optional Placeholder replaces default empty label
    ─ All popup frames parented to sg (ScreenGui root) for correct Z ordering
]]

local function Create(cls, props)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then o[k] = v end
    end
    if props and props.Parent then o.Parent = props.Parent end
    return o
end

local function Tween(obj, goals, ti)
    local tw = TweenService:Create(obj, ti or TI_MID, goals)
    tw:Play()
    return tw
end

local function Round(f, r)
    Create("UICorner", { CornerRadius = UDim.new(0, r or 8), Parent = f })
end

local function Stroke(f, col, thick)
    return Create("UIStroke", { Color = col or Theme.Border, Thickness = thick or 1, Parent = f })
end

local function Pad(f, t, b, l, r)
    Create("UIPadding", {
        PaddingTop    = UDim.new(0, t or 0),
        PaddingBottom = UDim.new(0, b or 0),
        PaddingLeft   = UDim.new(0, l or 0),
        PaddingRight  = UDim.new(0, r or 0),
        Parent = f,
    })
end

local function ListLayout(f, dir, gap)
    Create("UIListLayout", {
        FillDirection  = dir or Enum.FillDirection.Vertical,
        SortOrder      = Enum.SortOrder.LayoutOrder,
        Padding        = UDim.new(0, gap or 0),

local function ScrollFrame(parent, z)
    local sf = Create("ScrollingFrame", {
        Size                   = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,

local function IsMouseOrTouch(inputType)
    return inputType == Enum.UserInputType.MouseButton1
        or inputType == Enum.UserInputType.Touch
end

local function IsMouseMove(inputType)
    return inputType == Enum.UserInputType.MouseMovement
        or inputType == Enum.UserInputType.Touch
end

local function GetGuiParent()
    if isStudio then
        return LocalPlayer:FindFirstChild("PlayerGui")

    return CoreGui
end

-- Draggable: uses a single shared UserInputService.InputChanged
local function Draggable(handle, frame)
    local dr, ds, sp
    handle.InputBegan:Connect(function(i)
        if IsMouseOrTouch(i.UserInputType) then
            dr = true
            ds = i.Position
            sp = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dr and IsMouseMove(i.UserInputType) then
            local d = i.Position - ds
            frame.Position = UDim2.new(
                sp.X.Scale, sp.X.Offset + d.X,
                sp.Y.Scale, sp.Y.Offset + d.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if IsMouseOrTouch(i.UserInputType) then dr = false end
    end)
end

-- ─────────────────────────────────────────────────────────────
--  Serialise / Deserialise helpers for Config
-- ─────────────────────────────────────────────────────────────

local function SV(v)
    if typeof(v) == "Color3" then return { __type = "Color3", r = v.R, g = v.G, b = v.B }
    elseif type(v) == "table" then
        local o = {}
        for i, x in ipairs(v) do o[i] = SV(x) end
        return o
    end
    return v
end

local function DV(v)
    if type(v) == "table" then
        if v.__type == "Color3" then return Color3.new(v.r, v.g, v.b) end
        local o = {}
        for i, x in ipairs(v) do o[i] = DV(x) end
        return o
    end
    return v
end

local function CFG_DIR()  return "CrispyLib/" .. CrispyLib._configName end
local function CFG_PATH(n) return CFG_DIR() .. "/" .. n .. ".json" end
local function MKDIR()
    if _hasFolder then
        pcall(makefolder, "CrispyLib")
        pcall(makefolder, CFG_DIR())
    end
end

local function RegFlag(f, g, s)
    if f then CrispyLib._flagElements[f] = { Get = g, Set = s } end
end
local function SetFlag(f, v)
    if f ~= nil then CrispyLib._flags[f] = v end
end

-- ─────────────────────────────────────────────────────────────
--  Public: GetFlag
-- ─────────────────────────────────────────────────────────────
function CrispyLib.GetFlag(f)
    local e = CrispyLib._flagElements[f]
    if e then return e.Get() end
    return CrispyLib._flags[f]
end

-- ─────────────────────────────────────────────────────────────
--  Config API
-- ─────────────────────────────────────────────────────────────

CrispyLib.Config = {}

function CrispyLib.Config.Snapshot()
    local t = {}
    for f, e in pairs(CrispyLib._flagElements) do t[f] = SV(e.Get()) end
    return t
end

function CrispyLib.Config.Apply(snap)
    for f, v in pairs(snap) do
        local e = CrispyLib._flagElements[f]
        if e then pcall(e.Set, DV(v)) end
    end
end

function CrispyLib.Config.Export()
    local ok, j = pcall(function() return HttpService:JSONEncode(CrispyLib.Config.Snapshot()) end)
    return ok and j or "{}"
end

function CrispyLib.Config.Import(json)
    local ok, t = pcall(function() return HttpService:JSONDecode(json) end)
    if ok and type(t) == "table" then CrispyLib.Config.Apply(t); return true end
    return false
end

function CrispyLib.Config.Save(name)
    name = name or "default"
    if not _hasWrite then
        CrispyLib._mem = CrispyLib._mem or {}
        CrispyLib._mem[name] = CrispyLib.Config.Snapshot()
        return true
    end
    MKDIR()
    return pcall(writefile, CFG_PATH(name), CrispyLib.Config.Export())
end

function CrispyLib.Config.Load(name)
    name = name or "default"
    if not _hasRead then
        if CrispyLib._mem and CrispyLib._mem[name] then
            CrispyLib.Config.Apply(CrispyLib._mem[name]); return true
        end
        return false
    end
    local p = CFG_PATH(name)
    if _hasIsfile and not isfile(p) then return false end
    local ok, j = pcall(readfile, p)
    if not ok then return false end
    return CrispyLib.Config.Import(j)
end

function CrispyLib.Config.List()
    if not _hasList then
        local l = {}
        if CrispyLib._mem then for k in pairs(CrispyLib._mem) do l[#l + 1] = k end end
        return l
    end
    MKDIR()
    local ok, f = pcall(listfiles, CFG_DIR())
    if not ok then return {} end
    local n = {}
    for _, p in ipairs(f) do
        local nm = p:match("([^/\\]+)%.json$")
        if nm then n[#n + 1] = nm end
    end
    return n
end

function CrispyLib.Config.Delete(name)
    if not name then return false end
    if not _hasWrite then
        if CrispyLib._mem then CrispyLib._mem[name] = nil end
        return true
    end
    local p = CFG_PATH(name)
    if type(delfile) == "function" then pcall(delfile, p)
    else pcall(writefile, p, "{}") end
    return true
end

                elseif d:IsA("ImageLabel") then
                    pcall(function() Tween(d, { ImageTransparency = 1 }, TI_MID) end)
                elseif d:IsA("Frame") and d ~= bg then
                    pcall(function() Tween(d, { BackgroundTransparency = 1 }, TI_MID) end)
                end
            end
            task.delay(0.35, function()
                sg:Destroy()
                if cb then pcall(cb) end
            end)
        end)
    end

    -- animate loader bar pulse
    task.spawn(function()
        while sg and sg.Parent do
            Tween(barFill, { BackgroundColor3 = Color3.fromRGB(48, 168, 255) },
                TweenInfo.new(0.9, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut))
            task.wait(1)
            Tween(barFill, { BackgroundColor3 = Theme.LoaderBar },
                TweenInfo.new(0.9, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut))
            task.wait(1)
        end
    end)

    return L
end

        function Tab:AddToggle(tc)

            local knob = Create("Frame", {
                Size             = UDim2.new(0, 18, 0, 18),
                AnchorPoint      = Vector2.new(0, 0.5),
                Position         = UDim2.new(0, state and 23 or 3, 0.5, 0),
                BackgroundColor3 = Theme.ToggleKnob,
                BorderSizePixel  = 0,
                ZIndex           = Z.Content + 4,
                Parent           = tBg,
            })
            Round(knob, 9)
            local btn = Create("TextButton", {
                Size                   = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                Text                   = "",
                ZIndex                 = Z.Content + 5,
                Parent                 = tBg,
            })

            local function Upd()
                Tween(tBg,  { BackgroundColor3 = state and Theme.Accent or Theme.ToggleOff }, TI_MID)
                Tween(knob, { Position = UDim2.new(0, state and 23 or 3, 0.5, 0) }, TI_MID)
            end

            btn.MouseButton1Click:Connect(function()
                state = not state; Upd()
                SetFlag(tc.Flag, state)
                task.spawn(function() pcall(cb, state) end)
            end)
            SetFlag(tc.Flag, state)

            local obj = {}
            function obj:Set(v, s)
                state = v; Upd(); SetFlag(tc.Flag, state)
                if not s then task.spawn(function() pcall(cb, state) end) end
            end
            function obj:Get()           return state end
            function obj:UpdateName(n)   if nl then nl.Text = n end end  -- v3: fixed
            function obj:SetVisibility(s) row.Visible = s end
            function obj:UpdateState(v)  obj:Set(v) end
            function obj:Toggle()        obj:Set(not state) end
            RegFlag(tc.Flag, function() return state end, function(v) obj:Set(v, true) end)
            return obj
        end

                                Rebuild("")
                            end
                        end)

                    end
                end
            end

            local function OpenList()
                listOpen = true
                local abp = dBtn.AbsolutePosition
                local abs = dBtn.AbsoluteSize
                local listH = math.min(#options * 32 + (multi and 84 or 46) + 14, 240)
                listFrame.Position = UDim2.new(0, abp.X, 0, abp.Y + abs.Y + 4)
                listFrame.Visible  = true

                Tween(dStroke,   { Color = Theme.FocusBorder },          TI_MID)
            end

            dBtnClick.MouseButton1Click:Connect(function()
                if listOpen then CloseList() else OpenList() end
            end)

            -- outside-click guard (single connection, v3)
            TrackConn(UserInputService.InputBegan:Connect(function(inp)
                if not listOpen then return end
                if not IsMouseOrTouch(inp.UserInputType) then return end
                local mp  = UserInputService:GetMouseLocation()
                local fp  = listFrame.AbsolutePosition; local fs = listFrame.AbsoluteSize
                local dp2 = dBtn.AbsolutePosition;      local ds2 = dBtn.AbsoluteSize
                local insideList = mp.X >= fp.X and mp.X <= fp.X + fs.X and mp.Y >= fp.Y and mp.Y <= fp.Y + fs.Y
                local insideBtn  = mp.X >= dp2.X and mp.X <= dp2.X + ds2.X and mp.Y >= dp2.Y and mp.Y <= dp2.Y + ds2.Y
                if not insideList and not insideBtn then CloseList() end
            end))

            -- ── v3: Dropdown API ───────────────────────────────

            local obj = {}

            function obj:Get()         return selected end
            function obj:GetSelected() return selected end  -- alias

            function obj:Set(v, silent)
                if multi then
                    selected = type(v) == "table" and v or { v }
                else
                    selected = v
                end
                dLabel.Text = VT()
                SetFlag(dc.Flag, selected)
                if not silent then task.spawn(function() pcall(cb, selected) end) end
                Rebuild("")
            end

            -- v3: AddOption – append a new choice
            function obj:AddOption(opt)
                for _, v in ipairs(options) do
                    if v == opt then return end  -- already exists
                end
                options[#options + 1] = opt
                if not multi and (selected == "" or selected == placeholder) then
                    selected = opt; dLabel.Text = VT()
                end
                Rebuild(sIn.Text)
            end

            -- v3: RemoveOption – remove a choice; deselects if currently selected
            function obj:RemoveOption(opt)
                for i, v in ipairs(options) do
                    if v == opt then
                        table.remove(options, i)
                        break
                    end
                end
                if multi then
                    for i, v in ipairs(selected) do
                        if v == opt then table.remove(selected, i); break end
                    end
                elseif selected == opt then
                    selected = options[1] or ""
                end
                dLabel.Text = VT()
                SetFlag(dc.Flag, selected)
                Rebuild(sIn.Text)
            end

            -- v3: SetOptions – replace entire option list
            function obj:SetOptions(list)
                options = {}
                for _, v in ipairs(list) do options[#options + 1] = v end
                if multi then
                    -- keep only selected items that still exist
                    local newSel = {}
                    for _, v in ipairs(selected) do
                        for _, o in ipairs(options) do
                            if o == v then newSel[#newSel + 1] = v; break end
                        end
                    end
                    selected = newSel
                else
                    local found = false
                    for _, v in ipairs(options) do
                        if v == selected then found = true; break end
                    end
                    if not found then selected = options[1] or "" end
                end
                dLabel.Text = VT()
                SetFlag(dc.Flag, selected)
                Rebuild(sIn.Text)
            end

            -- v3: GetOptions – read the current option list
            function obj:GetOptions()
                local copy = {}
                for _, v in ipairs(options) do copy[#copy + 1] = v end
                return copy
            end

            function obj:UpdateName(n)   if nl then nl.Text = n end end
            function obj:SetVisibility(s) row.Visible = s end

            RegFlag(dc.Flag,
                function() return selected end,
                function(v) obj:Set(v, true) end)
            return obj
        end

        -- ── Tab:AddSlider ─────────────────────────────────────

        function Tab:AddSlider(sc)
            sc = sc or {}
            local min         = sc.Min or sc.Minimum or 0
            local max         = sc.Max or sc.Maximum or 100
            local suffix      = sc.Suffix     or ""
            local prefix      = sc.Prefix     or ""
            local step        = sc.Step       or 1
            local precision   = sc.Precision  or 0
            local cb          = sc.Callback   or function() end
            local onComplete  = sc.onInputComplete
            local val         = math.clamp(sc.Default or min, min, max)

            local function formatVal(v)
                if precision > 0 then
                    return prefix .. string.format("%." .. precision .. "f", v) .. suffix
                end
                return prefix .. tostring(v) .. suffix
            end

            local row = MkRow(self, 70)
            local nl  = RowLabels(row, sc.Name, sc.Description)

            local vLbl = Create("TextBox", {
                Name                   = "Val",
                Size                   = UDim2.new(0, 80, 0, 20),
                Position               = UDim2.new(1, -98, 0, 11),
                BackgroundColor3       = Theme.InputBg,
                BackgroundTransparency = 0.5,
                BorderSizePixel        = 0,
                Text                   = formatVal(val),
                TextColor3             = Theme.Accent,
                TextSize               = 13,
                FontFace               = InterBold(),
                TextXAlignment         = Enum.TextXAlignment.Right,
                ClearTextOnFocus       = true,
                ZIndex                 = Z.Content + 2,
                Parent                 = row,
            })
            Round(vLbl, 4)

            local trackBg = Create("Frame", {
                Size             = UDim2.new(1, -40, 0, 6),
                Position         = UDim2.new(0, 20, 1, -20),
                BackgroundColor3 = Theme.ToggleOff,
                BorderSizePixel  = 0,
                ZIndex           = Z.Content + 2,
                Parent           = row,
            })
            Round(trackBg, 4)
            local fill = Create("Frame", {
                Size             = UDim2.new((val - min) / (max - min), 0, 1, 0),
                BackgroundColor3 = Theme.Accent,
                BorderSizePixel  = 0,
                ZIndex           = Z.Content + 3,
                Parent           = trackBg,
            })
            Round(fill, 4)
            local knob = Create("Frame", {
                Size             = UDim2.new(0, 16, 0, 16),
                AnchorPoint      = Vector2.new(0.5, 0.5),
                Position         = UDim2.new((val - min) / (max - min), 0, 0.5, 0),
                BackgroundColor3 = Color3.fromRGB(255, 255, 255),
                BorderSizePixel  = 0,
                ZIndex           = Z.Content + 4,
                Parent           = trackBg,
            })
            Round(knob, 8)
            Create("UIStroke", { Color = Theme.Accent, Thickness = 2, Parent = knob })

            -- v3: cache AbsolutePosition/Size at drag-start
            local dr = false
            local dragABP, dragABS

            local function UpdSlider(posX)
                local t   = math.clamp((posX - dragABP) / dragABS, 0, 1)
                val = math.floor((min + t * (max - min)) / step + 0.5) * step
                val = math.clamp(val, min, max)
                local frac = (val - min) / (max - min)
                fill.Size       = UDim2.new(frac, 0, 1, 0)
                knob.Position   = UDim2.new(frac, 0, 0.5, 0)
                vLbl.Text       = formatVal(val)
                SetFlag(sc.Flag, val)
                task.spawn(function() pcall(cb, val) end)
            end

            trackBg.InputBegan:Connect(function(i)
                if IsMouseOrTouch(i.UserInputType) then
                    dr      = true
                    dragABP = trackBg.AbsolutePosition.X
                    dragABS = trackBg.AbsoluteSize.X
                    UpdSlider(i.Position.X)
                    Tween(knob, { Size = UDim2.new(0, 18, 0, 18) }, TI_FAST)
                end
            end)
            TrackConn(UserInputService.InputChanged:Connect(function(i)
                if dr and IsMouseMove(i.UserInputType) then UpdSlider(i.Position.X) end
            end))
            TrackConn(UserInputService.InputEnded:Connect(function(i)
                if IsMouseOrTouch(i.UserInputType) and dr then
                    if onComplete then task.spawn(function() pcall(onComplete, val) end) end
                    dr = false
                    Tween(knob, { Size = UDim2.new(0, 16, 0, 16) }, TI_FAST)
                end
            end))

            vLbl.FocusLost:Connect(function()
                local inputText   = vLbl.Text
                local numVal, isPct = inputText:match("^(%-?%d+%.?%d*)(%%?)$")
                if numVal then
                    numVal = tonumber(numVal)
                    if isPct == "%" then numVal = min + (numVal / 100) * (max - min) end
                    val = math.clamp(numVal, min, max)
                    val = math.floor(val / step + 0.5) * step
                end
                local frac = (val - min) / (max - min)
                fill.Size     = UDim2.new(frac, 0, 1, 0)
                knob.Position = UDim2.new(frac, 0, 0.5, 0)
                vLbl.Text     = formatVal(val)
                SetFlag(sc.Flag, val)
                task.spawn(function() pcall(cb, val) end)
                if onComplete then task.spawn(function() pcall(onComplete, val) end) end
            end)

            local obj = {}
            function obj:Set(v, s)
                val       = math.clamp(v, min, max)
                val       = math.floor(val / step + 0.5) * step
                local frac = (val - min) / (max - min)
                fill.Size     = UDim2.new(frac, 0, 1, 0)
                knob.Position = UDim2.new(frac, 0, 0.5, 0)
                vLbl.Text     = formatVal(val)
                SetFlag(sc.Flag, val)
                if not s then task.spawn(function() pcall(cb, val) end) end
            end
            function obj:Get()             return val end
            function obj:UpdateValue(v)    obj:Set(tonumber(v) or min) end
            function obj:GetValue()        return val end
            function obj:UpdateName(n)     if nl then nl.Text = n end end  -- v3: fixed
            function obj:SetVisibility(s)  row.Visible = s end
            RegFlag(sc.Flag, function() return val end, function(v) obj:Set(v, true) end)
            return obj
        end

                Size               = UDim2.new(1, -16, 0, 24),
                Position           = UDim2.new(0, 8, 0, rgbY),
                BackgroundTransparency = 1,
                ZIndex             = Z.Popup + 1,
                Parent             = pFrame,
            })
            ListLayout(hexFrame, Enum.FillDirection.Horizontal, 4)

            local function makeRGBInput(name, w)
                local bg = Create("Frame", {
                    Size             = UDim2.new(0, w, 0, 24),
                    BackgroundColor3 = Theme.InputBg,
                    BorderSizePixel  = 0,
                    ZIndex           = Z.Popup + 1,
                    Parent           = hexFrame,
                })
                Round(bg, 4); Stroke(bg, Theme.Border, 1)
                Create("TextLabel", {
                    Size               = UDim2.new(0, 14, 1, 0),
                    Position           = UDim2.new(0, 4, 0, 0),
                    BackgroundTransparency = 1,
                    Text               = name,
                    TextColor3         = Theme.SubtitleText,
                    TextSize           = 10,
                    FontFace           = InterSemiBold(),
                    ZIndex             = Z.Popup + 2,
                    Parent             = bg,
                })
                local box = Create("TextBox", {
                    Size               = UDim2.new(1, -20, 1, 0),
                    Position           = UDim2.new(0, 18, 0, 0),
                    BackgroundTransparency = 1,
                    Text               = "255",
                    TextColor3         = Theme.LabelText,
                    TextSize           = 11,
                    FontFace           = InterFont(),
                    TextXAlignment     = Enum.TextXAlignment.Left,
                    ClearTextOnFocus   = true,
                    ZIndex             = Z.Popup + 2,
                    Parent             = bg,
                })
                return box
            end

            local rBox   = makeRGBInput("R", 48)
            local gBox   = makeRGBInput("G", 48)
            local bBox   = makeRGBInput("B", 48)
            local hexBox = makeRGBInput("#", 68)

            local cPrev = Create("Frame", {
                Size                   = UDim2.new(1, -16, 0, 22),
                Position               = UDim2.new(0, 8, 0, rgbY + 30),
                BackgroundColor3       = val,
                BackgroundTransparency = alpha,
                BorderSizePixel        = 0,
                ZIndex                 = Z.Popup + 1,
                Parent                 = pFrame,
            })
            Round(cPrev, 6)

            local h, s, br = Color3.toHSV(val)

            local function UpdRGBBoxes()
                local r = math.floor(val.R * 255 + 0.5)
                local g = math.floor(val.G * 255 + 0.5)
                local b = math.floor(val.B * 255 + 0.5)
                rBox.Text   = tostring(r)
                gBox.Text   = tostring(g)
                bBox.Text   = tostring(b)
                hexBox.Text = string.format("%02X%02X%02X", r, g, b)
            end

            local function UpdColor()
                val                       = Color3.fromHSV(h, s, br)
                swatch.BackgroundColor3   = val
                swatch.BackgroundTransparency = alpha
                cPrev.BackgroundColor3    = val
                cPrev.BackgroundTransparency  = alpha
                local pureHue = Color3.fromHSV(h, 1, 1)
                sbBox.BackgroundColor3    = pureHue
                sbHueGrad.Color           = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
                    ColorSequenceKeypoint.new(1, pureHue),
                })
                sbKnob.Position = UDim2.new(s, -6, 1 - br, -6)
                hKnob.Position  = UDim2.new(h, -5, 0.5,    0)
                if hasAlpha and alphaKnob then
                    alphaKnob.Position          = UDim2.new(alpha, 0, 0.5, 0)
                    alphaBar.BackgroundColor3   = val
                end
                UpdRGBBoxes()
                SetFlag(cpc.Flag, val)
                task.spawn(function() pcall(cb, val, hasAlpha and alpha or nil) end)
            end
            UpdColor()

            -- v3: single shared drag-state for color picker
            local dSB, dH, dA = false, false, false

            local function SBDrag(pos)
                local a = sbBox.AbsolutePosition; local z = sbBox.AbsoluteSize
                s  = math.clamp((pos.X - a.X) / z.X, 0, 1)
                br = math.clamp(1 - (pos.Y - a.Y) / z.Y, 0, 1)
                UpdColor()
            end
            local function HDrag(pos)
                local a = hBar.AbsolutePosition; local z = hBar.AbsoluteSize
                h = math.clamp((pos.X - a.X) / z.X, 0, 1); UpdColor()
            end
            local function ADrag(pos)
                if not hasAlpha or not alphaBar then return end
                local a = alphaBar.AbsolutePosition; local z = alphaBar.AbsoluteSize
                alpha = math.clamp((pos.X - a.X) / z.X, 0, 1); UpdColor()
            end

            sbBtn.InputBegan:Connect(function(i)
                if IsMouseOrTouch(i.UserInputType) then dSB = true; SBDrag(i.Position) end
            end)
            hBtn.InputBegan:Connect(function(i)
                if IsMouseOrTouch(i.UserInputType) then dH = true; HDrag(i.Position) end
            end)
            if alphaBtn then
                alphaBtn.InputBegan:Connect(function(i)
                    if IsMouseOrTouch(i.UserInputType) then dA = true; ADrag(i.Position) end
                end)
            end

            TrackConn(UserInputService.InputChanged:Connect(function(i)
                if IsMouseMove(i.UserInputType) then
                    if dSB then SBDrag(i.Position) end
                    if dH  then HDrag(i.Position)  end
                    if dA  then ADrag(i.Position)  end
                end
            end))
            TrackConn(UserInputService.InputEnded:Connect(function(i)
                if IsMouseOrTouch(i.UserInputType) then dSB = false; dH = false; dA = false end
            end))

            local function updateFromRGB()
                local rv = math.clamp(tonumber(rBox.Text) or 0, 0, 255)
                local gv = math.clamp(tonumber(gBox.Text) or 0, 0, 255)
                local bv = math.clamp(tonumber(bBox.Text) or 0, 0, 255)
                rBox.Text = tostring(rv); gBox.Text = tostring(gv); bBox.Text = tostring(bv)
                h, s, br  = Color3.fromRGB(rv, gv, bv):ToHSV()
                UpdColor()
            end
            local function updateFromHex()
                local hex = hexBox.Text:gsub("#", "")
                if #hex == 6 then
                    local rv = tonumber(hex:sub(1, 2), 16) or 0
                    local gv = tonumber(hex:sub(3, 4), 16) or 0
                    local bv = tonumber(hex:sub(5, 6), 16) or 0
                    h, s, br = Color3.fromRGB(rv, gv, bv):ToHSV()
                    UpdColor()
                end
            end
            rBox.FocusLost:Connect(updateFromRGB)
            gBox.FocusLost:Connect(updateFromRGB)
            bBox.FocusLost:Connect(updateFromRGB)
            hexBox.FocusLost:Connect(updateFromHex)

            -- v3: outside-click guard is a single tracked connection
            TrackConn(UserInputService.InputBegan:Connect(function(inp)
                if not pOpen then return end
                if not IsMouseOrTouch(inp.UserInputType) then return end
                local mp = UserInputService:GetMouseLocation()
                local fp = pFrame.AbsolutePosition;  local fs = pFrame.AbsoluteSize
                local sp = swatch.AbsolutePosition; local ss = swatch.AbsoluteSize
                local insidePicker = mp.X >= fp.X and mp.X <= fp.X + fs.X and mp.Y >= fp.Y and mp.Y <= fp.Y + fs.Y
                local insideSwatch = mp.X >= sp.X and mp.X <= sp.X + ss.X and mp.Y >= sp.Y and mp.Y <= sp.Y + ss.Y
                if not insidePicker and not insideSwatch then
                    pOpen = false
                    Tween(pFrame, { Size = UDim2.new(0, CP_W, 0, 0) }, TI_MID)
                    task.delay(0.22, function() pFrame.Visible = false end)
                end
            end))

            local function ReposPicker()
                local abp = swatch.AbsolutePosition
                local abs2 = swatch.AbsoluteSize
                pFrame.Position = UDim2.new(0, abp.X - CP_W + abs2.X, 0, abp.Y + abs2.Y + 4)
            end

            swBtn.MouseButton1Click:Connect(function()
                pOpen = not pOpen
                if pOpen then
                    ReposPicker()
                    pFrame.Visible = true
                    Tween(pFrame, { Size = UDim2.new(0, CP_W, 0, CP_H) }, TI_MID)
                else
                    Tween(pFrame, { Size = UDim2.new(0, CP_W, 0, 0) }, TI_MID)
                    task.delay(0.22, function() pFrame.Visible = false end)
                end
            end)

            local obj = {}
            function obj:Set(color, silent)
                val = color; h, s, br = Color3.toHSV(color); UpdColor()
                if not silent then task.spawn(function() pcall(cb, val, hasAlpha and alpha or nil) end) end
            end
            function obj:Get()             return val end
            function obj:SetColor(c)       obj:Set(c) end
            function obj:SetAlpha(a)       alpha = a; UpdColor() end
            function obj:UpdateName(n)     if nl then nl.Text = n end end  -- v3: fixed
            function obj:SetVisibility(st) row.Visible = st end
            RegFlag(cpc.Flag, function() return val end, function(v) obj:Set(v, true) end)
            return obj
        end

        -- ── Tab:AddSeparator ──────────────────────────────────

        function Tab:AddSeparator(sc)
            sc = sc or {}
            self._order = self._order + 1
            if sc.Label then
                -- v3: separator with inline label
                local sf = Create("Frame", {
                    Name               = "Sep_" .. self._order,
                    Size               = UDim2.new(1, -24, 0, 20),
                    BackgroundTransparency = 1,
                    LayoutOrder        = self._order,
                    ZIndex             = Z.Content,
                    Parent             = self._content,
                })
                Create("Frame", {
                    Size             = UDim2.new(0.35, 0, 0, 1),
                    Position         = UDim2.new(0, 0, 0.5, 0),
                    BackgroundColor3 = Theme.Separator,
                    BorderSizePixel  = 0,
                    ZIndex           = Z.Content,
                    Parent           = sf,
                })
                Create("TextLabel", {
                    Size               = UDim2.new(0.3, 0, 1, 0),
                    Position           = UDim2.new(0.35, 0, 0, 0),
                    BackgroundTransparency = 1,
                    Text               = sc.Label,
                    TextColor3         = Theme.SectionLabel,
                    TextSize           = 10,
                    FontFace           = InterSemiBold(),
                    TextXAlignment     = Enum.TextXAlignment.Center,
                    ZIndex             = Z.Content + 1,
                    Parent             = sf,
                })
                Create("Frame", {
                    Size             = UDim2.new(0.35, 0, 0, 1),
                    Position         = UDim2.new(0.65, 0, 0.5, 0),
                    BackgroundColor3 = Theme.Separator,
                    BorderSizePixel  = 0,
                    ZIndex           = Z.Content,
                    Parent           = sf,
                })
            else
                Create("Frame", {
                    Name             = "Sep_" .. self._order,
                    Size             = UDim2.new(1, -24, 0, 1),
                    BackgroundColor3 = Theme.Separator,
                    BorderSizePixel  = 0,
                    LayoutOrder      = self._order,
                    ZIndex           = Z.Content,
                    Parent           = self._content,
                })
            end
        end

        function Tab:AddSpacer(h)
            self._order = self._order + 1
            Create("Frame", {
                Name               = "Spacer_" .. self._order,
                Size               = UDim2.new(1, 0, 0, h or 12),
                BackgroundTransparency = 1,
                LayoutOrder        = self._order,
                ZIndex             = Z.Content,
                Parent             = self._content,
            })
        end

        return Tab
    end  -- Win:AddTab

    -- ── Config Profiles tab ──────────────────────────────────

    if configName then
        Win:AddSidebarSection("Settings")
        local PT = Win:AddTab({ Name = "Profiles", Icon = "rbxassetid://0" })

        PT:AddSection({ Title = "Config Profiles", Description = "Save, load, import and export settings." })
        PT:AddSpacer(4)

        local nIn = PT:AddInput({ Name = "Profile Name", Description = "Name to save or load.",
            Default = "default", Placeholder = "e.g. myconfig" })

        PT:AddButton({ Name = "Save Config", Description = "Save current settings.", Label = "Save",
            Callback = function()
                local n = nIn:Get(); if n == "" then n = "default" end
                local ok = CrispyLib.Config.Save(n)
                CrispyLib.Notify({ Title = ok and "Saved" or "Failed",
                    Description = ok and ('Profile "' .. n .. '" saved.') or "Could not write.",
                    Type = ok and "success" or "error", Duration = 3 })
            end })

        local pList = CrispyLib.Config.List()
        if #pList == 0 then pList = { "default" } end
        local pDrop = PT:AddDropdown({ Name = "Load Profile", Description = "Choose saved profile.",
            Options = pList, Default = pList[1] })

        PT:AddButton({ Name = "Load Config", Description = "Apply selected profile.", Label = "Load",
            Callback = function()
                local n  = tostring(pDrop:Get())
                local ok = CrispyLib.Config.Load(n)
                CrispyLib.Notify({ Title = ok and "Loaded" or "Not Found",
                    Description = ok and ('Profile "' .. n .. '" applied.') or ('Profile "' .. n .. '" not found.'),
                    Type = ok and "success" or "error", Duration = 3 })
            end })

        PT:AddButton({ Name = "Refresh List", Description = "Reload profile names.", Label = "Refresh",
            Callback = function()
                local l = CrispyLib.Config.List(); if #l == 0 then l = { "default" } end
                pDrop:SetOptions(l)
                CrispyLib.Notify({ Title = "Refreshed", Description = #l .. " profile(s) found.",
                    Type = "info", Duration = 2 })
            end })

        PT:AddButton({ Name = "Delete Profile", Description = "Delete selected profile.", Label = "Delete",
            Style = "Danger",
            Callback = function()
                local n = tostring(pDrop:Get())
                CrispyLib.Config.Delete(n)
                CrispyLib.Notify({ Title = "Deleted",
                    Description = 'Profile "' .. n .. '" removed.', Type = "warn", Duration = 3 })
            end })

        PT:AddSeparator({ Label = "Export / Import" })
        PT:AddSpacer(4)
        PT:AddSection({ Title = "Export / Import", Description = "Share configs as JSON strings." })
        PT:AddSpacer(4)

        local expBox = PT:AddInput({ Name = "Exported JSON", Description = "Click Export to populate.",
            Default = "", Placeholder = "JSON will appear here..." })
        PT:AddButton({ Name = "Export Config", Description = "Serialise settings to JSON.", Label = "Export",
            Callback = function()
                expBox:Set(CrispyLib.Config.Export(), true)
                CrispyLib.Notify({ Title = "Exported", Description = "JSON in Export field.",
                    Type = "info", Duration = 3 })
            end })

        local impBox = PT:AddInput({ Name = "Import JSON", Description = "Paste JSON then click Import.",
            Default = "", Placeholder = "Paste JSON here..." })
        PT:AddButton({ Name = "Import Config", Description = "Apply from pasted JSON.", Label = "Import",
            Callback = function()
                local ok = CrispyLib.Config.Import(impBox:Get())
                CrispyLib.Notify({ Title = ok and "Imported" or "Invalid JSON",
                    Description = ok and "Settings applied." or "Could not parse JSON.",
                    Type = ok and "success" or "error", Duration = 3 })
            end })

        if autoLoad then task.defer(function() CrispyLib.Config.Load("default") end) end
    end

    return Win
end
